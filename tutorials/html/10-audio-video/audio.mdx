import {useEffect, useRef, useState} from 'react';

export function makeToneUrl() {
  // Builds a short three-note WAV file in memory, so the demos need no audio file.
  const rate = 11025;
  const notes = [440, 554, 659];
  const noteSeconds = 0.4;
  const samplesPerNote = Math.floor(rate * noteSeconds);
  const total = samplesPerNote * notes.length;
  const buffer = new ArrayBuffer(44 + total * 2);
  const view = new DataView(buffer);
  const write = (offset, text) => {
    for (let i = 0; i < text.length; i++) view.setUint8(offset + i, text.charCodeAt(i));
  };
  write(0, "RIFF");
  view.setUint32(4, 36 + total * 2, true);
  write(8, "WAVE");
  write(12, "fmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, rate, true);
  view.setUint32(28, rate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  write(36, "data");
  view.setUint32(40, total * 2, true);
  let index = 0;
  notes.forEach((frequency) => {
    for (let i = 0; i < samplesPerNote; i++) {
      const fade = Math.min(1, i / 300, (samplesPerNote - i) / 300);
      const value = Math.sin((2 * Math.PI * frequency * i) / rate) * 0.35 * fade;
      view.setInt16(44 + index * 2, value * 32767, true);
      index++;
    }
  });
  return URL.createObjectURL(new Blob([buffer], {type: "audio/wav"}));
}

export function formatTime(seconds) {
  if (!isFinite(seconds)) return "0.0";
  return seconds.toFixed(1);
}

export function AudioLab() {
  const audioRef = useRef(null);
  const [url, setUrl] = useState(null);
  const [loop, setLoop] = useState(false);
  const [state, setState] = useState({paused: true, time: 0, duration: 0, volume: 1, events: []});
  useEffect(() => {
    const made = makeToneUrl();
    setUrl(made);
    return () => URL.revokeObjectURL(made);
  }, []);
  const note = (name) => setState((old) => {
    const audio = audioRef.current;
    return {
      paused: audio ? audio.paused : old.paused,
      time: audio ? audio.currentTime : old.time,
      duration: audio && isFinite(audio.duration) ? audio.duration : old.duration,
      volume: audio ? audio.volume : old.volume,
      events: name ? [...old.events.slice(-4), name] : old.events,
    };
  });
  return (
    <div>
      <p style={{margin: "0 0 0.5rem"}}>
        <label>
          <input type="checkbox" checked={loop} onChange={(event) => setLoop(event.target.checked)} /> loop
        </label>
      </p>
      {url === null ? (
        <p role="status">Preparing the sound…</p>
      ) : (
        <audio
          ref={audioRef}
          src={url}
          controls
          loop={loop}
          preload="metadata"
          onLoadedMetadata={() => note("loadedmetadata")}
          onPlay={() => note("play")}
          onPause={() => note("pause")}
          onEnded={() => note("ended")}
          onVolumeChange={() => note("volumechange")}
          onTimeUpdate={() => note(null)}
        >
          Your browser cannot play this sound.
        </audio>
      )}
      <ul style={{margin: "0.6rem 0 0.3rem", paddingLeft: "1.25rem"}}>
        <li><code>paused</code>: {String(state.paused)}</li>
        <li><code>currentTime</code>: {formatTime(state.time)} s of {formatTime(state.duration)} s</li>
        <li><code>volume</code>: {state.volume.toFixed(2)}</li>
        <li>Latest events: {state.events.length ? state.events.join(", ") : "none yet"}</li>
      </ul>
      <p style={{margin: 0, fontSize: "0.9em", color: "#64748b"}}>
        This sound is generated by the page itself (three short notes), so no file is needed. Press play, drag the bar, change the volume, and watch the values change.
      </p>
    </div>
  );
}

export function AttributeBuilder() {
  const [opts, setOpts] = useState({controls: true, loop: false, muted: false, autoplay: false});
  const [preload, setPreload] = useState("metadata");
  const toggle = (key) => setOpts((old) => ({...old, [key]: !old[key]}));
  const parts = ['src="episode.mp3"'];
  if (opts.controls) parts.push("controls");
  if (opts.loop) parts.push("loop");
  if (opts.muted) parts.push("muted");
  if (opts.autoplay) parts.push("autoplay");
  if (preload !== "metadata") parts.push('preload="' + preload + '"');
  const code = "<audio " + parts.join(" ") + "></audio>";
  const notes = [];
  if (!opts.controls && !opts.autoplay) notes.push({bad: true, text: "No controls and no script: the visitor sees nothing and cannot play it."});
  if (!opts.controls && opts.autoplay) notes.push({bad: true, text: "Sound that starts by itself and cannot be paused is a serious accessibility problem."});
  if (opts.autoplay && !opts.muted) notes.push({bad: true, text: "Autoplay with sound is blocked by most browsers, and annoys people when it works."});
  if (opts.loop && !opts.controls) notes.push({bad: true, text: "A sound that loops and cannot be stopped is hard to bear."});
  if (preload === "auto") notes.push({bad: false, text: "preload=\"auto\" downloads the whole file early. It uses data even if nobody presses play."});
  if (preload === "none") notes.push({bad: false, text: "preload=\"none\" saves data. The first press of play may take a moment."});
  if (notes.length === 0) notes.push({bad: false, text: "A sensible, polite player."});
  return (
    <form onSubmit={(event) => event.preventDefault()}>
      <fieldset style={{margin: "0 0 0.5rem"}}>
        <legend>Boolean attributes</legend>
        {Object.keys(opts).map((key) => (
          <label key={key} style={{display: "inline-block", marginRight: "1rem"}}>
            <input type="checkbox" checked={opts[key]} onChange={() => toggle(key)} /> {key}
          </label>
        ))}
      </fieldset>
      <p style={{margin: "0 0 0.5rem"}}>
        <label htmlFor="ab-preload">preload </label>
        <select id="ab-preload" value={preload} onChange={(event) => setPreload(event.target.value)}>
          <option value="none">none</option>
          <option value="metadata">metadata (default in most browsers)</option>
          <option value="auto">auto</option>
        </select>
      </p>
      <pre style={{margin: "0 0 0.5rem", whiteSpace: "pre-wrap"}}><code>{code}</code></pre>
      <ul role="status" style={{margin: 0, paddingLeft: "1.25rem"}}>
        {notes.map((entry) => (
          <li key={entry.text} style={{color: entry.bad ? "#b91c1c" : "#15803d"}}>{entry.text}</li>
        ))}
      </ul>
    </form>
  );
}

Sound can make a page feel alive: a podcast episode, a pronunciation guide, a song sample, or a short voice message. The **`<audio>`** element adds it with one line of HTML, and the browser supplies a complete player, with play, pause, a progress bar, and volume.

This lesson shows how to add sound well: which attributes to use, which formats to choose, how the autoplay rules work, and how to make audio useful for people who cannot hear it.

The demo above is a **real** `<audio>` element, playing a short sound that the page builds itself. Try the player, and watch the values underneath.

Three things to notice:

1. **`src`** names the file, like the `src` of an image.
2. **`controls`** shows the browser's player. Without it, nothing appears on the page.
3. **The text inside** is **fallback content.** A browser that supports `<audio>` ignores it, and one that does not shows it. A download link is the most useful fallback.

The player looks different in every browser. That is normal, and each one is already keyboard and screen reader friendly.

## Attributes

| Attribute | What it does | Advice |
| :-- | :-- | :-- |
| `src` | The address of the audio file | Or use `source` elements for several formats (see [Source](./source)) |
| `controls` | Shows the player | **Almost always include it** |
| `loop` | Starts again when it ends | Only with controls, and rarely |
| `muted` | Starts with the sound off | Needed for any autoplay attempt |
| `autoplay` | Tries to start by itself | **Avoid** (see below) |
| `preload` | How much to download early: `none`, `metadata`, or `auto` | Use `none` or `metadata` to save data |
| `crossorigin` | How to fetch a file from another site | Needed for some features, such as analysing the sound |

### `preload`

| Value | Meaning | Good for |
| :-- | :-- | :-- |
| `none` | Download nothing until the visitor presses play | A page with many players |
| `metadata` | Download only the length and basic details | The usual choice |
| `auto` | Let the browser download as it likes, often the whole file | A single, important track |

`preload` is a **hint**, not an order. The browser may ignore it, especially on mobile connections where it tries to save data.

## Choosing a Format

A browser plays a file only if it understands its **format** (the container and the codec inside). Common audio formats:

| Format | Typical file | Notes |
| :-- | :-- | :-- |
| **MP3** | `.mp3` | The safest choice. Plays almost everywhere, with good quality at modest size |
| **AAC** | `.m4a`, `.aac` | Good quality at small size, widely supported |
| **Ogg Vorbis or Opus** | `.ogg`, `.opus` | Efficient and free of licences. Support varies by browser, so check before relying on it |
| **WAV** | `.wav` | Uncompressed, so files are large. Good for short sound effects |
| **FLAC** | `.flac` | Lossless and large. Support is common now, but check |

For most websites, **one MP3 or AAC file** is enough. If you need to be sure, offer two formats with `<source>`, and let the browser choose. See [Source](./source).

Format support can change over time, so test in the browsers your visitors use.

:::tip Keep files small
Speech does not need music quality. A podcast episode at a lower bitrate sounds fine and loads much faster, which matters to visitors on slow or expensive connections.
:::

## Autoplay: Please Don't

`autoplay` asks the browser to start playing as soon as it can. It sounds convenient, but:

- **Most browsers block it** when the audio has sound, unless the visitor has already interacted with the page. The attribute silently does nothing.
- When it **does** work, it startles people, clashes with screen reader speech, and annoys anyone in a quiet place.
- It wastes data for visitors who never wanted the sound.

If you really need sound after a click, start it **from the click handler**:

```js
document.getElementById("start").addEventListener("click", () => {
  document.getElementById("sound").play();
});
```

A click counts as the visitor's request, so the browser allows it.

## Controlling Audio with JavaScript

Every `<audio>` element has a built-in API:

```js
const audio = document.querySelector("audio");

audio.play();               // returns a Promise
audio.pause();
audio.currentTime = 30;     // jump to 30 seconds
audio.volume = 0.5;         // 0 (silent) to 1 (full)
audio.playbackRate = 1.5;   // faster
audio.muted = true;

audio.addEventListener("ended", () => console.log("Finished"));
```

`play()` returns a **promise**, which fails if the browser blocks the sound. Always handle that:

```js
audio.play().catch((error) => {
  console.log("Playback was blocked:", error.name);
  // Show a visible "Play" button instead
});
```

The useful events are `loadedmetadata`, `play`, `pause`, `ended`, `timeupdate`, `volumechange`, and `error`. You saw some of them in the demo's event list. Building your own controls is covered in [Media Controls](./media-controls).

## Transcripts: Sound for Everyone

An audio file is **invisible to search engines, to people who cannot hear it, and to anyone in a place where they cannot play it.** A **transcript** (the words, written down) fixes all of that.

```html
<figure>
  <figcaption>Episode 12: Why bread rises</figcaption>
  <audio controls src="episode-12.mp3"></audio>
</figure>

<details>
  <summary>Read the transcript</summary>
  <p><strong>Asha:</strong> Welcome back to the show…</p>
</details>
```

- Put the transcript **on the same page,** or link to it right next to the player.
- Include **who is speaking** and meaningful sounds, such as "[laughter]".
- Use a [details and summary](./details) widget if the text is long, so it does not take over the page.
- Captions with `<track>` are for **video**. For audio-only content, a transcript is the standard way. See [Accessible Media](./accessible-media).

## Accessibility at a Glance

- **Always provide `controls`,** or your own accessible controls.
- **Never autoplay sound,** and never play sound that cannot be paused.
- **Add a transcript** for speech, and describe important non-speech sounds.
- **Label the player** with a nearby heading or `figcaption`, or `aria-label="Episode 12"`.
- **Do not rely on sound alone** for instructions or errors. Add a visual message.
- **Keep the sound reasonable.** Avoid sudden loud sounds.
- **Test with the keyboard:** the player must be reachable with Tab, and Space must play and pause.

## Security and Privacy Notes

- **Audio files can always be downloaded.** Do not rely on hiding the player, or on a "no download" setting, to protect them. If sound must be private, serve it only to signed-in users, on the server.
- **Own the rights.** Music, voice recordings, and sound effects are usually protected by copyright.
- **Use HTTPS.** An insecure audio file on a secure page can be blocked as mixed content.
- **Be careful with user-uploaded audio.** Check the file type and size on the server, and never use the file name straight in HTML. Escape it first.
- **Third-party players** can load trackers. Check what an embed loads before adding it, and ask for consent where required.

## Global Audience Notes

- **Provide a transcript in the language of the audio,** and a translation where you can, so non-native listeners can read along.
- **Mark the language** of the transcript with `lang` if it differs from the page.
- **Offer slower playback.** The built-in player often has a speed menu, which helps language learners.
- **Think about data costs.** Use `preload="none"` or `metadata`, compress files, and tell people the size ("Download, 12 MB").
- **Be careful with idioms and humour** in audio, which translate badly, and add clarity in the transcript.
- **Respect quiet spaces.** Many people read in public places, so sound must always be the visitor's choice.

## Common Mistakes to Avoid

1. **Leaving out `controls`.**
2. **Using `autoplay`,** especially with sound.
3. **Looping music that cannot be stopped.**
4. **No fallback content or download link.**
5. **No transcript.**
6. **A huge uncompressed WAV** for a long recording.
7. **Using only a format some browsers cannot play,** and not testing.
8. **`preload="auto"` on a page with many players,** which downloads them all.
9. **Not handling the rejected promise** from `play()`.
10. **Assuming the audio is private** because the player is hidden.
11. **Using sound alone** to show an error or a success.
12. **No label** to say what the audio is.

## Key Takeaways

- `<audio controls src="…">` adds a full, accessible player with one line of HTML.
- Always include **`controls`**, and give a **fallback** with a download link.
- Choose **MP3 or AAC** for safety, keep files small, and use `preload="none"` or `metadata`.
- **Do not autoplay.** Start sound only from a visitor's action, and handle the promise from `play()`.
- Use the JavaScript API (`play`, `pause`, `currentTime`, `volume`, events) for custom behaviour.
- Add a **transcript** for every piece of speech.
- Hidden audio is not private, and embedded players can track visitors.
- Plan for slow connections, many languages, and quiet places.
