export function AriaBehaviourLab() {
  const [tabbable, setTabbable] = useState(false);
  const [keys, setKeys] = useState(false);
  const empty = {focus: 0, clicks: 0, enter: 0, space: 0, actions: 0};
  const [fake, setFake] = useState(empty);
  const [real, setReal] = useState(empty);
  const bump = (setter, field, extra) => setter((previous) => ({...previous, [field]: previous[field] + 1, actions: previous.actions + (extra ? 1 : 0)}));
  const onFakeKey = (event) => {
    if (event.key === "Enter") bump(setFake, "enter", keys);
    if (event.key === " ") {
      if (keys) event.preventDefault();
      bump(setFake, "space", keys);
    }
  };
  const onRealKey = (event) => {
    if (event.key === "Enter") setReal((previous) => ({...previous, enter: previous.enter + 1}));
    if (event.key === " ") setReal((previous) => ({...previous, space: previous.space + 1}));
  };
  const box = {border: "1px solid #94a3b8", borderRadius: "8px", padding: "0.7rem", flex: "1 1 14rem"};
  return (
    <div>
      <fieldset style={{margin: "0 0 0.7rem"}}>
        <legend>Help the made-up button along (the real one needs nothing)</legend>
        <label style={{display: "block"}}><input type="checkbox" checked={tabbable} onChange={(event) => setTabbable(event.target.checked)} /> Add <code>tabindex="0"</code> so Tab can reach it</label>
        <label style={{display: "block"}}><input type="checkbox" checked={keys} onChange={(event) => setKeys(event.target.checked)} /> Write script so Enter and Space work</label>
      </fieldset>
      <div style={{display: "flex", gap: "0.8rem", flexWrap: "wrap", marginBottom: "0.7rem"}}>
        <div style={box}>
          <p style={{margin: "0 0 0.4rem"}}><strong>A div with role="button"</strong></p>
          <div role="button" tabIndex={tabbable ? 0 : undefined} onFocus={() => bump(setFake, "focus", false)} onClick={() => bump(setFake, "clicks", true)} onKeyDown={onFakeKey} style={{display: "inline-block", padding: "0.4rem 0.9rem", border: "2px solid #1d4ed8", borderRadius: "6px", cursor: "pointer"}}>Save</div>
          <ul style={{margin: "0.5rem 0 0", paddingLeft: "1.2rem"}}>
            <li>Reached by Tab: <strong>{fake.focus}</strong></li>
            <li>Mouse clicks: <strong>{fake.clicks}</strong></li>
            <li>Enter received: <strong>{fake.enter}</strong></li>
            <li>Space received: <strong>{fake.space}</strong></li>
            <li>Times it did its job: <strong>{fake.actions}</strong></li>
          </ul>
        </div>
        <div style={box}>
          <p style={{margin: "0 0 0.4rem"}}><strong>A real button</strong></p>
          <button type="button" onFocus={() => bump(setReal, "focus", false)} onClick={() => bump(setReal, "clicks", true)} onKeyDown={onRealKey}>Save</button>
          <ul style={{margin: "0.5rem 0 0", paddingLeft: "1.2rem"}}>
            <li>Reached by Tab: <strong>{real.focus}</strong></li>
            <li>Clicks (mouse, Enter, or Space): <strong>{real.clicks}</strong></li>
            <li>Enter received: <strong>{real.enter}</strong></li>
            <li>Space received: <strong>{real.space}</strong></li>
            <li>Times it did its job: <strong>{real.actions}</strong></li>
          </ul>
        </div>
      </div>
      <p style={{margin: "0 0 0.5rem"}}><button type="button" onClick={() => { setFake(empty); setReal(empty); }}>Reset the counts</button></p>
      <p style={{margin: 0, fontSize: "0.9em", color: "#64748b"}}>
        Press Tab to move through the page, and try Enter and Space on each one. To a screen reader, both are "button" (that is what <code>role="button"</code> does). But the made-up one starts with no keyboard support at all: Tab skips it, and Enter and Space do nothing. ARIA changed what it is called, not what it does. You have to add the focus and the key handling yourself, and then you have rebuilt, in more code, what the real button gave you for free. (Counts are for this demo only. Some screen readers have their own ways to activate items, so they may behave differently.)
      </p>
    </div>
  );
}

export function AriaCheckerInner() {
  const [which, setWhich] = useState("messy");
  const [html, setHtml] = useState(ariaSamples.messy);
  const doc = new DOMParser().parseFromString("<!doctype html><body>" + html, "text/html");
  const focusSelector = 'a[href], button:not([disabled]), input:not([disabled]):not([type="hidden"]), select:not([disabled]), textarea:not([disabled]), summary, [tabindex]:not([tabindex="-1"])';
  const elements = Array.from(doc.querySelectorAll("*")).filter((el) => el.hasAttribute("role") || Array.from(el.attributes).some((attr) => attr.name.indexOf("aria-") === 0));
  const widgetRoles = ["button", "link", "checkbox", "switch", "tab", "menuitem", "radio", "option", "slider"];
  const nativeTags = ["button", "a", "input", "select", "textarea", "summary"];
  const rows = elements.map((el) => {
    const tag = el.tagName.toLowerCase();
    const issues = [];
    const notes = [];
    const role = (el.getAttribute("role") || "").trim().toLowerCase().split(/\s+/)[0];
    const focusable = el.matches(focusSelector);
    if (role) {
      if (knownRoles.indexOf(role) === -1) issues.push("Unknown role \"" + role + "\". Check the spelling.");
      else {
        const implicit = tag === "a" ? (el.hasAttribute("href") ? "link" : "") : implicitRoles[tag];
        if (implicit === role) notes.push("Redundant: a " + tag + " already has the role " + role + ".");
        if (widgetRoles.indexOf(role) !== -1 && nativeTags.indexOf(tag) === -1 && !focusable) issues.push("role=\"" + role + "\" on a " + tag + " that cannot be focused. ARIA adds no keyboard behaviour. Use a native element, or add tabindex and key handling.");
        if ((role === "presentation" || role === "none") && focusable) issues.push("A focusable element must not have role=" + role + ".");
      }
    }
    Array.from(el.attributes).forEach((attr) => {
      if (attr.name.indexOf("aria-") !== 0) return;
      const name = attr.name.slice(5);
      const value = attr.value.trim().toLowerCase();
      if (knownAttrs.indexOf(name) === -1) {
        issues.push("Unknown attribute " + attr.name + ". Check the spelling.");
        return;
      }
      if (allowedValues[name] && allowedValues[name].indexOf(value) === -1) issues.push(attr.name + "=\"" + attr.value + "\" is not a valid value. Allowed: " + allowedValues[name].join(", ") + ".");
      if (idRefAttrs.indexOf(name) !== -1) {
        attr.value.split(/\s+/).filter(Boolean).forEach((id) => { if (!doc.getElementById(id)) issues.push(attr.name + " points to the id \"" + id + "\", which does not exist."); });
      }
    });
    if (el.getAttribute("aria-hidden") === "true" && (focusable || el.querySelector(focusSelector))) issues.push("aria-hidden=\"true\" on something focusable. Keyboard users can reach it, but screen reader users cannot see it.");
    if ((el.hasAttribute("aria-label") || el.hasAttribute("aria-labelledby")) && !role && /^(div|span|p|b|i|em|strong|small|u)$/.test(tag)) notes.push("A name on a plain " + tag + " is not reliably announced. Give it a role, or put the text in the content.");
    if (el.hasAttribute("aria-pressed") && tag !== "button" && role !== "button") issues.push("aria-pressed belongs on a button.");
    if (el.hasAttribute("aria-checked") && tag === "input" && /^(checkbox|radio)$/.test((el.getAttribute("type") || "").toLowerCase())) notes.push("A native checkbox or radio already has a checked state. Remove aria-checked, or it can disagree with the real state.");
    return {tag, role, issues, notes};
  });
  const bad = rows.reduce((sum, row) => sum + row.issues.length, 0);
  const choose = (value) => { setWhich(value); setHtml(ariaSamples[value]); };
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>Which HTML do you want to check?</legend>
        <label style={{marginRight: "1rem"}}><input type="radio" name="aria-sample" checked={which === "messy"} onChange={() => choose("messy")} /> Messy ARIA</label>
        <label><input type="radio" name="aria-sample" checked={which === "fixed"} onChange={() => choose("fixed")} /> The fixed version</label>
      </fieldset>
      <p style={{margin: "0 0 0.5rem"}}>
        <label htmlFor="aria-html">HTML to check (edit it freely)</label><br />
        <textarea id="aria-html" rows={10} value={html} onChange={(event) => setHtml(event.target.value)} style={{width: "100%", fontFamily: "monospace"}} />
      </p>
      <p style={{margin: "0 0 0.4rem"}}><strong>{rows.length}</strong> elements use ARIA, with <strong>{bad}</strong> problems.</p>
      <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
        {rows.map((row, index) => (
          <li key={index} style={{marginBottom: "0.4rem"}}>
            <code>{"<" + row.tag + (row.role ? " role=" + row.role : "") + ">"}</code>{" "}
            {row.issues.length === 0 ? <strong>OK</strong> : <span><strong>Problem:</strong> {row.issues.join(" ")}</span>}
            {row.notes.length > 0 ? <span> <em>Note:</em> {row.notes.join(" ")}</span> : null}
          </li>
        ))}
      </ul>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        A teaching checker with a shortened list of roles and attributes. It cannot know whether your ARIA is the right ARIA, or whether a widget really works. Use a full checker (such as your browser's accessibility panel, axe, or Lighthouse) and test with real assistive technology.
      </p>
    </div>
  );
}

export function AriaChecker() {
  return <BrowserOnly>{() => <AriaCheckerInner />}</BrowserOnly>;
}

So far in this section you have used plain HTML to make pages accessible: real headings, real buttons, real labels, real tables. That is the best foundation, and it will carry you a long way. But sometimes HTML alone is not enough. A tab panel, a live "Saved!" message, an icon-only button: these need a little extra help to tell assistive technology what is going on.

That help is **ARIA.** It is powerful, and easy to misuse. This lesson explains what ARIA is, what it can and **cannot** do, and the few rules that keep you out of trouble. The next lessons look at the main parts one by one.

## What Is ARIA?

**ARIA** stands for **Accessible Rich Internet Applications.** (You will also see **WAI-ARIA,** from the W3C's Web Accessibility Initiative.) It is a set of **attributes** you add to HTML elements to give assistive technology more information:

- **Roles** say **what** something is (`role="tablist"`).
- **Properties** describe **it** (`aria-label="Close"`, `aria-controls="menu"`).
- **States** say **what it is doing right now** (`aria-expanded="true"`, `aria-pressed="false"`).

Think of ARIA as **a note stuck on the accessibility tree** (see [Introduction to Accessibility](./introduction-to-a11y)). It changes what a screen reader **hears,** and nothing else.

### What ARIA Does Not Do

- **It does not add behaviour.** `role="button"` makes something sound like a button. It does not make it focusable, or give it Enter and Space support, as the lab showed.
- **It does not change how things look.** A `div` with `role="heading"` is not bigger or bolder.
- **It does not fix the focus order** or the keyboard.
- **It does not work magic on its own.** Support varies between browsers and screen readers, and a wrong role can make things **worse** than none.

:::warning
You may hear **"No ARIA is better than bad ARIA."** Large surveys of home pages (such as the WebAIM Million) have repeatedly found that pages **with** ARIA tend to have **more** detected accessibility errors than pages without it. That does not mean ARIA is bad. It means ARIA is **easy to get wrong,** and a wrong role is worse than a missing one. Use it with care, and test.
:::

## The Five Rules of ARIA

The W3C's guide to using ARIA boils everything down to five rules:

1. **If you can use a native HTML element or attribute with the meaning and behaviour you need, use it,** instead of adding ARIA. (Use `button`, not `div role="button"`.)
2. **Do not change native semantics,** unless you really have to. (Do not write `<h2 role="button">`. Put a `button` inside the heading.)
3. **All interactive ARIA controls must be usable with the keyboard.** If you make a `role="button"`, it must be focusable and respond to Enter and Space.
4. **Do not use `role="presentation"` or `aria-hidden="true"` on a focusable element.** It would be reachable by keyboard, but invisible to a screen reader.
5. **All interactive elements must have an accessible name.** See [Accessible Names](./accessible-names).

If you remember only one thing, remember rule 1.

## Native HTML First

Almost every ARIA role you meet has a **native** equivalent that is better:

| Instead of this | Use this |
| :-- | :-- |
| `div role="button"` | `button` |
| `div role="link"` | `a href` |
| `div role="navigation"` | `nav` |
| `div role="main"` | `main` |
| `div role="heading" aria-level="2"` | `h2` |
| `div role="list"` and `listitem` | `ul` and `li` |
| `div role="checkbox"` | `input type="checkbox"` |
| `div role="dialog"` | `dialog` |
| `aria-required="true"` | `required` |
| `aria-disabled` on a form field | `disabled` |

Native elements come with **focus, keyboard support, a role, a name, and a state,** all tested by browser makers for years. ARIA is for the gaps where HTML has **no** native answer.

## Roles, Properties, and States

Here is a map of the ARIA you will meet. Each group gets its own lesson soon.

| Kind | What it does | Examples |
| :-- | :-- | :-- |
| **Roles** | Say what something is | `tablist`, `dialog`, `status`, `alert`, `search` |
| **Properties** | Describe and connect things | `aria-label`, `aria-labelledby`, `aria-describedby`, `aria-controls` |
| **States** | Change as people use the page | `aria-expanded`, `aria-pressed`, `aria-checked`, `aria-selected`, `aria-hidden`, `aria-current` |
| **Live regions** | Announce changes | `aria-live`, `role="status"`, `aria-atomic` |

### How the Syntax Works

- Every ARIA attribute begins with **`aria-`,** and the role is the plain **`role`** attribute.
- **True and false are words:** `aria-expanded="true"` and `aria-expanded="false"`. Write them as strings, with the quotes.
- Some values come from a **fixed list:** `aria-live="polite"`, `aria-current="page"`, `aria-pressed="mixed"`.
- Some values are the **`id` of another element:** `aria-labelledby="title"`, `aria-controls="menu"`. The id must exist.
- Some are **plain text:** `aria-label="Close"`. Write it in the page's language.
- **States must be kept up to date with script.** If a panel opens and `aria-expanded` still says `false`, ARIA is now lying to people.
- A **missing** attribute and a `false` value can mean different things for some attributes (for example `aria-pressed`: not a toggle at all, or a toggle that is off). Follow the rules for each attribute.

## Where to Learn the Patterns

When you must build a custom widget (tabs, a menu, a combobox, an accordion), do **not** invent the ARIA yourself. Follow a tested pattern, such as those in the **WAI-ARIA Authoring Practices Guide (APG).** It lists the roles, states, and keyboard keys for each widget, with examples.

Even a good pattern needs **testing,** because screen reader support differs. A pattern that works well in one combination (say, a screen reader and browser on Windows) may behave differently on a phone. Test with the combinations your audience really uses.

## Good Habits

1. **Reach for native HTML first,** every time.
2. **Add ARIA only to fill a real gap.**
3. **Keep ARIA states up to date** with script.
4. **Make every ARIA control work with a keyboard.**
5. **Never hide a focusable element** with `aria-hidden`.
6. **Follow a tested pattern** for custom widgets.
7. **Check that every id you reference exists.**
8. **Write names in the page's language,** and keep them short.
9. **Test with real assistive technology,** and review ARIA in code review.

## Accessibility Checklist

- [ ] No ARIA is used where a native element or attribute would do.
- [ ] Every ARIA role is valid, and suits the element.
- [ ] Every custom widget can be focused and used with a keyboard.
- [ ] No focusable element has `aria-hidden="true"` or `role="presentation"`.
- [ ] Every interactive element has an accessible name.
- [ ] ARIA states (`aria-expanded`, `aria-pressed`, and so on) change when the widget changes.
- [ ] Every id used in `aria-labelledby`, `aria-describedby`, and `aria-controls` exists.
- [ ] ARIA values are valid, and spelled correctly.
- [ ] Custom widgets follow a known pattern, and were tested with a screen reader.
- [ ] The ARIA has been checked with an automated tool, and by a person.

## Security and Privacy Notes

- **Do not put private data in ARIA attributes.** Values such as `aria-label` can be read aloud, read by browser extensions, and seen in the page source. Do not include passwords, personal identifiers, or hidden information.
- **Do not use ARIA to mislead.** A name that does not match what a control really does (for example "Cancel" on a button that confirms a purchase) is deceptive, and harms people who rely on it.
- **Be careful what you announce.** A live region that reads out sensitive information (a one-time code, an account balance, a typed password) could be overheard by someone nearby.
- **Do not use `aria-hidden` to hide important warnings,** security messages, or legal notices from screen reader users.
- **Do not detect or track assistive technology use** by looking for ARIA behaviour or settings. It can reveal a disability.
- **Test that ARIA cannot be abused** if you render content from users. Never insert user-supplied text into `role` or `aria-*` attributes without cleaning it first.

## Global Audience Notes

- **Translate ARIA text.** Values such as `aria-label`, `aria-description`, and `aria-roledescription` are part of your content. Include them in your translation process, and test with the tools your audience uses.
- **Set the page language** with `lang`, so screen readers pick the right voice for the names ARIA supplies.
- **Avoid `aria-roledescription`** unless you have a clear need. A custom description is read in your words, and may be wrong in other languages, or confuse people who know the usual role names.
- **Right-to-left languages:** arrow-key patterns (tabs, menus, sliders) should follow the reading direction. Test them in a right-to-left layout.
- **Assistive technology differs by region and device.** Many people use a screen reader on a phone, and some use older or free tools. Test on both a desktop and a mobile screen reader, and keep your markup simple.
- **Simple wording** in ARIA labels translates best, and is easier for people reading in a second language.

## Common Mistakes to Avoid

1. **`div` or `span` with `role="button"`** and no keyboard support.
2. **Putting ARIA on everything,** "just in case."
3. **Adding `role="button"` to a `button`,** or `role="navigation"` to a `nav`.
4. **`aria-hidden="true"` on a focusable element.**
5. **States that never change,** such as `aria-expanded` stuck on `false`.
6. **Pointing to ids that do not exist.**
7. **Misspelled roles and attributes** (`aria-lable`).
8. **Invalid values** (`aria-expanded="yes"`).
9. **Using `aria-label` on a plain `div` or `span`.**
10. **Using ARIA to repair bad HTML** instead of fixing the HTML.
11. **Inventing your own widget patterns.**
12. **Never testing with a screen reader.**

## Quick Recap

<details>
<summary>1. What does ARIA stand for, and what does it do?</summary>

Accessible Rich Internet Applications. It adds roles, properties, and states that tell assistive technology more about an element.

</details>

<details>
<summary>2. What is the first rule of ARIA?</summary>

If a native HTML element or attribute can do the job, use it, and do not add ARIA.

</details>

<details>
<summary>3. Does `role="button"` make a `div` work like a button?</summary>

No. It only changes what assistive technology announces. You still have to add focus, and Enter and Space support, yourself.

</details>

<details>
<summary>4. What are the three kinds of ARIA attributes?</summary>

Roles (what it is), properties (what it is like, and what it relates to), and states (what it is doing right now).

</details>

<details>
<summary>5. Why must you never put `aria-hidden="true"` on a focusable element?</summary>

Keyboard users could still reach it, but screen reader users would not know it exists.

</details>

<details>
<summary>6. When do you need ARIA?</summary>

When HTML has no native answer: custom widgets such as tabs, live messages, extra states such as `aria-expanded`, and missing names for icon-only controls.

</details>

## Key Takeaways

- **ARIA adds meaning, not behaviour.** It changes what assistive technology hears, and nothing else.
- **Native HTML first.** It is shorter, tested, and more reliable.
- **Roles, properties, and states** are the three parts of ARIA.
- **Follow the five rules,** especially: keyboard support, no hiding focusable things, and an accessible name for everything interactive.
- **Keep states up to date,** and make sure referenced ids exist.
- **Copy tested patterns** for custom widgets, and do not invent your own.
- **Test with a keyboard and a screen reader.** Bad ARIA is worse than none.
