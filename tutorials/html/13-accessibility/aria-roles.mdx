export function RoleExplorer() {
  const [category, setCategory] = useState("All");
  const [query, setQuery] = useState("");
  const [picked, setPicked] = useState("button");
  const list = roleData.filter((item) => (category === "All" || item.cat === category) && (query.trim() === "" || item.role.indexOf(query.trim().toLowerCase()) !== -1 || item.native.toLowerCase().indexOf(query.trim().toLowerCase()) !== -1));
  const current = roleData.find((item) => item.role === picked) || list[0] || roleData[0];
  return (
    <div>
      <div style={{display: "flex", gap: "1rem", flexWrap: "wrap", marginBottom: "0.6rem"}}>
        <p style={{margin: 0}}>
          <label htmlFor="role-category">Category</label><br />
          <select id="role-category" value={category} onChange={(event) => setCategory(event.target.value)}>
            {roleCategories.map((name) => (<option key={name} value={name}>{name}</option>))}
          </select>
        </p>
        <p style={{margin: 0}}>
          <label htmlFor="role-query">Search by role or element</label><br />
          <input id="role-query" type="search" value={query} onChange={(event) => setQuery(event.target.value)} />
        </p>
      </div>
      <p style={{margin: "0 0 0.4rem"}}><strong>{list.length}</strong> roles shown. Choose one to see the details.</p>
      <div role="group" aria-label="Roles" style={{display: "flex", gap: "0.3rem", flexWrap: "wrap", marginBottom: "0.7rem"}}>
        {list.map((item) => (
          <button key={item.role} type="button" aria-pressed={current.role === item.role} onClick={() => setPicked(item.role)} style={{padding: "0.2rem 0.6rem", fontWeight: current.role === item.role ? 700 : 400, background: current.role === item.role ? "#dbeafe" : undefined, color: current.role === item.role ? "#1e293b" : undefined, border: current.role === item.role ? "2px solid #1d4ed8" : undefined}}>{item.role}</button>
        ))}
      </div>
      <div style={{border: "1px solid #94a3b8", borderRadius: "8px", padding: "0.7rem"}} aria-live="polite">
        <p style={{margin: "0 0 0.3rem"}}><strong>{current.role}</strong> ({current.cat})</p>
        <p style={{margin: "0 0 0.3rem"}}><strong>Native HTML:</strong> {current.native}</p>
        <p style={{margin: "0 0 0.3rem"}}><strong>Needs a name:</strong> {current.name}</p>
        <p style={{margin: "0 0 0.3rem"}}><strong>Keyboard:</strong> {current.keys}</p>
        <p style={{margin: "0 0 0.5rem"}}>{current.note}</p>
        <pre style={{margin: 0, overflowX: "auto"}}><code>{current.code}</code></pre>
      </div>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This is a selection of the most useful roles, not the whole list. For a custom widget, always follow a tested pattern, because a role also brings rules about names, keyboard keys, required children, and states.
      </p>
    </div>
  );
}

export function LandmarkLab() {
  const [semantic, setSemantic] = useState(true);
  const [nameSecond, setNameSecond] = useState(false);
  const [useSearch, setUseSearch] = useState(true);
  const [useAside, setUseAside] = useState(true);
  const [regionMode, setRegionMode] = useState("none");
  const landmarks = [];
  const problems = [];
  if (semantic) {
    landmarks.push("banner");
    landmarks.push("navigation: Main");
    landmarks.push(nameSecond ? "navigation: Footer links" : "navigation");
    if (!nameSecond) problems.push("Two navigation landmarks, and one has no name. Someone browsing landmarks cannot tell them apart.");
    if (useSearch) landmarks.push("search: Recipes");
    landmarks.push("main");
    if (useAside) landmarks.push("complementary: Related recipes");
    if (regionMode === "named") landmarks.push("region: Delivery");
    if (regionMode === "unnamed") problems.push("A section with no name is not a landmark at all, so it is missing from the list.");
    landmarks.push("contentinfo");
  } else {
    problems.push("Only div elements, so there are no landmarks. Someone using a landmark list finds nothing, and must read the whole page in order.");
  }
  const lines = [];
  lines.push(semantic ? "<header>…</header>" : "<div class=\"header\">…</div>");
  lines.push(semantic ? "<nav aria-label=\"Main\">…</nav>" : "<div class=\"menu\">…</div>");
  if (useSearch) lines.push(semantic ? "<form role=\"search\" aria-label=\"Recipes\">…</form>" : "<div class=\"search\">…</div>");
  lines.push(semantic ? "<main>" : "<div class=\"content\">");
  if (regionMode !== "none") lines.push(semantic ? (regionMode === "named" ? "  <section aria-labelledby=\"delivery\"><h2 id=\"delivery\">Delivery</h2>…</section>" : "  <section>…</section>") : "  <div class=\"box\">…</div>");
  lines.push(semantic ? "</main>" : "</div>");
  if (useAside) lines.push(semantic ? "<aside aria-label=\"Related recipes\">…</aside>" : "<div class=\"side\">…</div>");
  lines.push(semantic ? "<nav" + (nameSecond ? " aria-label=\"Footer links\"" : "") + ">…</nav>" : "<div class=\"links\">…</div>");
  lines.push(semantic ? "<footer>…</footer>" : "<div class=\"footer\">…</div>");
  return (
    <div>
      <fieldset style={{margin: "0 0 0.7rem"}}>
        <legend>Build the page</legend>
        <label style={{display: "block"}}><input type="checkbox" checked={semantic} onChange={(event) => setSemantic(event.target.checked)} /> Use landmark elements (header, nav, main, footer)</label>
        <label style={{display: "block"}}><input type="checkbox" checked={nameSecond} onChange={(event) => setNameSecond(event.target.checked)} disabled={!semantic} /> Name the second nav ("Footer links")</label>
        <label style={{display: "block"}}><input type="checkbox" checked={useSearch} onChange={(event) => setUseSearch(event.target.checked)} /> Add a search area</label>
        <label style={{display: "block"}}><input type="checkbox" checked={useAside} onChange={(event) => setUseAside(event.target.checked)} /> Add a sidebar (aside)</label>
        <p style={{margin: "0.4rem 0 0"}}>
          <label htmlFor="landmark-region">A section for delivery</label>{" "}
          <select id="landmark-region" value={regionMode} onChange={(event) => setRegionMode(event.target.value)}>
            <option value="none">None</option>
            <option value="unnamed">A section with no name</option>
            <option value="named">A section with a name</option>
          </select>
        </p>
      </fieldset>
      <p style={{margin: "0 0 0.3rem"}}><strong>The HTML:</strong></p>
      <pre style={{margin: "0 0 0.7rem", overflowX: "auto"}}><code>{lines.join("\n")}</code></pre>
      <div style={{border: "1px solid #94a3b8", borderRadius: "8px", padding: "0.7rem", marginBottom: "0.6rem"}} aria-live="polite">
        <p style={{margin: "0 0 0.3rem"}}><strong>A screen reader's landmark list:</strong></p>
        {landmarks.length === 0 ? <p style={{margin: 0}}><em>(no landmarks found)</em></p> : (
          <ol style={{margin: 0, paddingLeft: "1.2rem"}}>
            {landmarks.map((item, index) => (<li key={index}>{item}</li>))}
          </ol>
        )}
      </div>
      {problems.length > 0 ? (
        <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
          {problems.map((item, index) => (<li key={index}><strong>Problem:</strong> {item}</li>))}
        </ul>
      ) : <p style={{margin: 0}}><strong>Looks good:</strong> every landmark is there, and every repeated one has its own name.</p>}
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        Screen readers can list the landmarks of a page, or jump from one to the next, so people can skip straight to the main content or the search. The names and the list shown here are a simplified example of that, and each screen reader words it a little differently. Also note that the header and footer are page-level landmarks only when they sit directly in the page, and not inside an article or section.
      </p>
    </div>
  );
}

export function LiveRolesLab() {
  const [statusText, setStatusText] = useState("");
  const [alertText, setAlertText] = useState("");
  const [logLines, setLogLines] = useState([]);
  const [transcript, setTranscript] = useState([]);
  const note = (line) => setTranscript((previous) => previous.concat(line).slice(-7));
  const save = () => {
    if (statusText === "Recipe saved.") note("status: the text did not change, so nothing new is announced.");
    else {
      setStatusText("Recipe saved.");
      note("status: \"Recipe saved.\" Announced politely, when the screen reader is not busy.");
    }
  };
  const fail = () => {
    if (alertText === "Could not save. Check your connection.") note("alert: the text did not change, so nothing new is announced.");
    else {
      setAlertText("Could not save. Check your connection.");
      note("alert: \"Could not save...\" Announced at once, and may interrupt what is being read.");
    }
  };
  const chat = () => {
    const line = "Message " + (logLines.length + 1) + " from Sam";
    setLogLines(logLines.concat(line));
    note("log: \"" + line + "\" Added to the end, and announced politely.");
  };
  const reset = () => {
    setStatusText("");
    setAlertText("");
    setLogLines([]);
    setTranscript([]);
  };
  const box = {border: "1px solid #94a3b8", borderRadius: "8px", padding: "0.5rem 0.7rem", marginBottom: "0.5rem", minHeight: "2.4rem"};
  return (
    <div>
      <p style={{margin: "0 0 0.6rem", display: "flex", gap: "0.4rem", flexWrap: "wrap"}}>
        <button type="button" onClick={save}>Save recipe</button>
        <button type="button" onClick={fail}>Make an error</button>
        <button type="button" onClick={chat}>Add a chat line</button>
        <button type="button" onClick={reset}>Reset</button>
      </p>
      <p style={{margin: "0 0 0.2rem"}}><strong>role="status"</strong> (polite)</p>
      <div role="status" style={box}>{statusText}</div>
      <p style={{margin: "0 0 0.2rem"}}><strong>role="alert"</strong> (assertive)</p>
      <div role="alert" style={box}>{alertText}</div>
      <p style={{margin: "0 0 0.2rem"}}><strong>role="log"</strong> (a history)</p>
      <div role="log" aria-label="Chat" style={box}>
        {logLines.map((line, index) => (<p key={index} style={{margin: 0}}>{line}</p>))}
      </div>
      <p style={{margin: "0.4rem 0 0.2rem"}}><strong>What a screen reader would do (simulation):</strong></p>
      {transcript.length === 0 ? <p style={{margin: 0}}><em>Press a button.</em></p> : (
        <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
          {transcript.map((line, index) => (<li key={index}>{line}</li>))}
        </ul>
      )}
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        These are real live regions, so if you use a screen reader, you will hear them. The list above is only a plain-text simulation of the difference. Two things matter: the region must already be in the page before its text changes, and an identical message is not announced again. The exact behaviour depends on the screen reader and browser, so test it. Do not use <code>alert</code> for ordinary messages, because it interrupts people.
      </p>
    </div>
  );
}

A **role** tells assistive technology **what an element is.** "This is a button." "This is a navigation area." "This is a tab." It is the first thing a screen reader tells you about something, and it decides how a person expects to use it: a button is pressed, a checkbox is toggled, a link is followed.

The key point, which you met in the last lesson, is that **native HTML elements already have roles.** A `button` is a button, a `nav` is a navigation landmark, an `h2` is a level 2 heading. The `role` attribute is for the situations where HTML has no element that fits.

## How Roles Work

- **Every element has a role,** either **implicit** (from the HTML element) or **explicit** (from the `role` attribute).
- **An explicit role wins** over the implicit one. `<div role="button">` is a button to assistive technology, even though it is a `div` in the code.
- **A role is a promise.** It tells people what to expect: a `button` should respond to Enter and Space, a `tab` should respond to arrow keys, a `dialog` should keep focus inside. The role does **not** keep the promise for you. You must write the behaviour.
- **Each role has rules.** Some need a **name** (`dialog`, `img`), some need **children** of a certain kind (`tablist` needs `tab`, `list` needs `listitem`), and some need **states** (`checkbox` needs `aria-checked` if it is custom).
- **A role needs to fit the element.** The "ARIA in HTML" rules list which roles are allowed on which elements. For example, you may not put `role="button"` on a heading, or on a link, without changing what the element is.
- **The `role` attribute can hold a list** of words, and browsers use the first one they understand. It is rarely needed, and support is not something to rely on, so use one role.
- **Roles are read by the browser's accessibility tree,** which you saw in [Introduction to Accessibility](./introduction-to-a11y).

## The Families of Roles

| Family | What it covers | Examples |
| :-- | :-- | :-- |
| **Landmark** | Big areas of the page | `banner`, `navigation`, `main`, `search`, `contentinfo` |
| **Widget** | Things you interact with | `button`, `checkbox`, `tab`, `slider`, `combobox` |
| **Live region** | Messages that change | `status`, `alert`, `log`, `timer` |
| **Document structure** | The shape of content | `list`, `heading`, `img`, `table`, `group` |
| **Window** | Pop-up windows | `dialog`, `alertdialog` |

There is also a group of **abstract** roles used to define the others. **Never** use them in your pages.

## Landmark Roles

Landmarks mark the **big areas** of a page. Screen reader users can list them, or jump between them, much as sighted users glance at the layout. They are the single biggest help for people who need to get to the main content, the search box, or the menu without reading everything in order.

You almost never need the `role` attribute for them, because HTML elements do the job:

| Element | Landmark role |
| :-- | :-- |
| `header` (directly in the page) | `banner` |
| `nav` | `navigation` |
| `main` | `main` |
| `aside` | `complementary` |
| `footer` (directly in the page) | `contentinfo` |
| `section` (with a name) | `region` |
| `form` (with a name) | `form` |
| `search`, or `form role="search"` | `search` |

Rules of thumb:

- **One `main`** per page, and use `header` and `footer` for the page (not for sections).
- **Name repeated landmarks,** with `aria-label` or `aria-labelledby`: two `nav` elements need different names.
- **Do not repeat the role in the name:** "Main," not "Main navigation."
- **Use `section` and `region` sparingly.** Too many landmarks are as unhelpful as none.
- **Put everything inside a landmark.** Content outside any landmark is easy to miss.
- **A newer `search` element exists** in recent HTML. Support is growing, so check your target browsers, and use `role="search"` on a form as the safe option.

## Widget Roles

Widget roles describe **interactive** things. Native elements cover many of them (`button`, `checkbox`, `radio`, `textbox`, `slider`, `progressbar`, `option`). A few have **no native element,** so you need roles to build them: `tab`, `tablist`, `tabpanel`, `switch`, `menu`, `menuitem`, `tooltip`, and a full `combobox`.

Using a widget role is a **contract.** It means you will provide:

1. **A name.**
2. **The right states** (for example `aria-selected` on a tab).
3. **The keyboard behaviour** people expect (arrow keys inside tabs and menus).
4. **Focus handling** (a roving `tabindex` in a tab list, for instance).
5. **The right structure** (tabs inside a tablist).

If you cannot provide all five, **do not use the role.** A half-built widget confuses people more than a plain one.

:::warning
**`menu` is not for website navigation.** A `role="menu"` tells a screen reader to switch into a special mode where the arrow keys are the way to move, like an application menu. A site's main menu is a list of links, so use `nav`, `ul`, and `a`.
:::

## Live Region Roles

Some roles tell assistive technology to **announce changes** to a region, **without moving focus.** The page must already contain the region, and then you change the text in it.

| Role | Announced | Use it for |
| :-- | :-- | :-- |
| `status` | Politely (when the reader is free) | "Recipe saved," "3 results found" |
| `alert` | At once, and may interrupt | Urgent errors, "Session ending" |
| `log` | Politely, new items added | Chat, activity history |
| `timer` | Not announced as it changes | A countdown, with a name |

Try them in the lab. If you use a screen reader, you will hear the real thing.

## Structure and Special Roles

Most structure roles come from HTML: `ul` and `ol` (`list`), `li` (`listitem`), `h1` to `h6` (`heading`), `table`, `img`, `figure`, `hr` (`separator`), `article`. A few explicit roles are worth knowing:

- **`role="img"`** makes a group of things (an SVG, a few emoji) read as **one image,** with a name from `aria-label`.
- **`role="group"`** with a name gathers related controls, such as a set of toolbar buttons. For form fields, prefer `fieldset` and `legend`.
- **`role="presentation"` or `role="none"`** removes the role of an element. It is useful to turn a **layout table** into plain content, or to hide a `li` inside a custom widget. **Never put it on something focusable or interactive.**
- **`role="list"` on a `ul` with `list-style: none`.** Some browser and screen reader combinations (notably Safari with VoiceOver) have stopped treating such a list as a list. Adding the role puts it back. This behaviour has changed over time, so check what your audience's browsers do before adding it everywhere.

## Roles to Avoid

- **Abstract roles** (such as `widget`, `landmark`, `command`). They are only for the ARIA specification.
- **`role="application"`.** It tells screen readers to hand over **all** the keys to your page, and turn off their own navigation. Almost no page needs it.
- **`role="menu"` for site navigation.** Use a `nav` with links.
- **A role that repeats the element:** `<button role="button">`, `<nav role="navigation">`. It does nothing useful, and clutters your code.
- **A role that fights the element:** `<h2 role="button">`, `<a role="heading">`. Wrap the content instead: `<h2><button>…</button></h2>`.
- **`role="alert"` for everyday messages.**
- **Interactive roles with no keyboard support** (see [Accessible Buttons](./accessible-buttons)).
- **Roles you do not understand.** If you cannot say what a role promises, do not use it.

## Check Your Roles

1. **Open the Accessibility panel** in your browser's DevTools, and look at the **Role** of each element. Does it match what you intended?
2. **List the landmarks** with a screen reader, or a browser extension that shows them. Is there one `main`? Are repeated ones named?
3. **Search your code for `role=`.** For each one, ask: is there a native element? Is the role allowed on this element? Do I provide the name, states, and keys that go with it?
4. **Run an automated checker** (axe, Lighthouse) for invalid roles, missing required children, and missing names.
5. **Use the keyboard** on every widget role, and listen with a screen reader.
6. **Check the "ARIA in HTML" rules** if you are unsure which roles an element may have.

## Good Habits

1. **Use the element with the right role,** and add `role` only to fill a gap.
2. **Use landmark elements,** and name the repeated ones.
3. **Treat a widget role as a promise** of name, states, keys, and focus.
4. **Follow a tested pattern** for tabs, menus, combo boxes, and dialogs.
5. **Choose `status` before `alert`.**
6. **Never put `presentation`, `none`, or `aria-hidden` on focusable elements.**
7. **Keep roles in line with the real behaviour.**
8. **Test with a screen reader,** and review roles in code review.

## Accessibility Checklist

- [ ] No role is used where a native element does the job.
- [ ] The page has one `main`, and a `header` and `footer`.
- [ ] Repeated landmarks have unique names.
- [ ] Every widget role has a name, the right states, and the right keyboard behaviour.
- [ ] Required parent and child roles are present (such as `tablist` with `tab`).
- [ ] `dialog` and `alertdialog` have names, and move focus in and out.
- [ ] `status` is used for normal messages, and `alert` is rare.
- [ ] `menu` is not used for site navigation, and `application` is not used.
- [ ] No focusable element has `role="presentation"` or `role="none"`.
- [ ] Roles do not repeat or fight the element they are on.

## Security and Privacy Notes

- **Do not announce sensitive data in live regions.** A `status` or `alert` that reads out an account number, a one-time code, or a password can be overheard.
- **Do not use roles to disguise controls.** A control that looks harmless but is announced as something else, or the other way round, is a deceptive pattern, and can hide dangerous actions from people who cannot see the screen.
- **Alerts must be honest.** A fake "Security warning" in an `alert` role, used to push people into clicking, is a scam technique.
- **Be careful with user-supplied text** in roles and ARIA values. Never insert unchecked user input into attributes.
- **Role-based detection** of screen reader users (by looking for how they use landmarks or live regions) could reveal a disability. Avoid tracking it.
- **Dialogs that ask for consent or privacy choices** need proper roles and focus, so that everyone can answer. Hidden or hard-to-reach "reject" buttons are a dark pattern.

## Global Audience Notes

- **Landmark names are text,** so translate them (`aria-label="Main"`, `aria-label="Footer links"`) along with the rest of the page.
- **Do not repeat the role name** in your label. Screen readers say "navigation" in the user's own language, so "Main navigation" can become a repeated word after translation.
- **Set `lang` on the page,** and on any labels in other languages.
- **Right-to-left languages:** arrow-key behaviour in tabs, menus, and toolbars should follow the reading direction. Test it.
- **Live regions in other languages:** make sure the message text is in the page's language, and short enough to be understood when read aloud.
- **Different devices:** many people use a screen reader on a phone. Roles are understood there too, but some patterns (such as menus and combo boxes) behave differently on touch. Test on mobile.

## Common Mistakes to Avoid

1. **`div role="button"` with no keyboard support.**
2. **`role="button"` on a `button`,** or `role="navigation"` on a `nav`.
3. **A `section` with no name,** expected to be a landmark.
4. **Two `nav` elements with no names.**
5. **`role="menu"` for a site navigation.**
6. **`role="application"` "to be safe."**
7. **`role="alert"` for every message.**
8. **A `tablist` with no `tab` children.**
9. **`role="presentation"` on something focusable.**
10. **Landmarks nested or repeated without reason.**
11. **Content outside any landmark.**
12. **Never checking the roles in the Accessibility panel.**

<details>
<summary>1. What is a role?</summary>

It tells assistive technology what an element is, such as a button, a navigation area, or a tab, and so how a person should expect to use it.

</details>

<details>
<summary>2. What is the difference between an implicit and an explicit role?</summary>

An implicit role comes from the HTML element (a `nav` is a navigation). An explicit role is set with the `role` attribute, and it wins over the implicit one.

</details>

<details>
<summary>3. Does adding `role="tab"` to a `button` make it behave like a tab?</summary>

No. The role is a promise. You must add the states (`aria-selected`), the keyboard behaviour (arrow keys), and the focus handling yourself.

</details>

<details>
<summary>4. Why name two `nav` landmarks differently?</summary>

So a person browsing the landmark list can tell them apart, for example "Main" and "Footer links."

</details>

<details>
<summary>5. When should you use `role="status"` and when `role="alert"`?</summary>

Use `status` for ordinary updates, announced politely. Use `alert` rarely, for urgent messages that must be announced at once.

</details>

<details>
<summary>6. Why is `role="menu"` wrong for site navigation?</summary>

It switches screen readers to an application-style mode with arrow-key navigation. A website menu is a list of links, so use `nav`, `ul`, and `a`.

</details>

## Key Takeaways

- **A role says what something is.** Native elements already have the right ones.
- **Landmark elements** (`header`, `nav`, `main`, `aside`, `footer`) are the easiest big win. Name the repeated ones.
- **A widget role is a promise** of a name, states, keys, and focus. Keep it, or do not use the role.
- **Live regions** need a role, and must already exist in the page. Prefer `status`, and use `alert` rarely.
- **Avoid** abstract roles, `application`, `menu` for navigation, and roles that repeat or fight the element.
- **Never** put `presentation`, `none`, or `aria-hidden` on focusable elements.
- **Check roles** in the Accessibility panel, and test with a screen reader.
