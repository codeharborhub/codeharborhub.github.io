export function OrderLab() {
  const [mode, setMode] = useState("classic");
  const [log, setLog] = useState([]);
  const frame = useRef(null);
  const current = loadModes.find((item) => item.key === mode);
  useEffect(() => {
    const handler = (event) => {
      if (!frame.current || event.source !== frame.current.contentWindow) return;
      if (event.data && typeof event.data.order === "string") {
        setLog((previous) => previous.concat(event.data.order));
      }
    };
    window.addEventListener("message", handler);
    return () => window.removeEventListener("message", handler);
  }, []);
  const external = encodeURIComponent("log('B: external script (" + current.label + ")')");
  const page = "<!DOCTYPE html><html><head><script>function log(m){parent.postMessage({order:m},'*')}document.addEventListener('DOMContentLoaded',function(){log('DOMContentLoaded fired')});addEventListener('load',function(){log('load fired')});log('A: inline script in the head')</script>"
    + "<script" + current.attrs + " src=\"data:text/javascript," + external + "\"></script>"
    + "</head><body><p>Hello</p><script>log('C: inline script, start of body')</script><p>More content</p><script>log('D: inline script, end of body')</script></body></html>";
  const explain = {
    classic: "A classic script stops the parser. B runs right where it is written, before the body is even read.",
    defer: "A deferred script downloads without stopping the parser, and runs after the page is read, just before DOMContentLoaded.",
    async: "An async script runs as soon as it has arrived, so it can land anywhere. With a tiny inline address like this one, it is usually early. On a real network, it could be much later.",
    module: "A module script is deferred by default. It runs after the page is read, in order, just before DOMContentLoaded.",
  };
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>How script B is loaded (it sits in the head)</legend>
        {loadModes.map((item) => (
          <label key={item.key} style={{display: "inline-block", marginRight: "1rem"}}>
            <input type="radio" name="load-mode-lab" checked={mode === item.key} onChange={() => { setMode(item.key); setLog([]); }} /> <code>{item.label}</code>
          </label>
        ))}
      </fieldset>
      <iframe ref={frame} key={mode} title="A real page running scripts" srcDoc={page} sandbox="allow-scripts" style={{width: "100%", height: "5rem", border: "1px solid #94a3b8"}} />
      <p style={{margin: "0.5rem 0 0.2rem"}}><strong>The order in which things really happened:</strong></p>
      <ol role="status" style={{margin: 0}}>
        {log.length === 0 ? <li>(waiting for the page to run)</li> : log.map((entry, index) => (<li key={index}>{entry}</li>))}
      </ol>
      <p style={{margin: "0.5rem 0 0"}}>{explain[mode]}</p>
      <p style={{margin: "0.5rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This is a real page that really runs scripts, inside a sandboxed frame, and it reports back with messages. Script B is supplied as a tiny inline address so no files are needed. Real files take longer to arrive, which makes the difference between the modes bigger. If this documentation site restricts scripts, the log may stay empty.
      </p>
    </div>
  );
}

  '<script src="/js/both.js" defer async></script>',
  '<script src="https://cdn.example.com/lib.js" integrity="sha384-abc"></script>',
  '<script src="http://example.com/old.js" defer></script>',
  '<script type="text/javascript">document.write("hi")</script>',

export function ScriptChecker() {
  const [source, setSource] = useState(scriptSample);
  const doc = new DOMParser().parseFromString("<!DOCTYPE html><html><head>" + source + "</head><body></body></html>", "text/html");
  const scripts = Array.from(doc.head.querySelectorAll("script"));
  const rows = scripts.map((script) => {
    const src = script.getAttribute("src");
    const type = (script.getAttribute("type") || "").toLowerCase();
    const isModule = type === "module";
    const isData = type !== "" && type !== "module" && type !== "text/javascript" && type !== "application/javascript";
    const notes = [];
    if (isData) {
      notes.push("A data block (" + type + "). The browser does not run it, which is correct for JSON-LD or other data.");
    } else {
      if (src && !isModule && !script.hasAttribute("defer") && !script.hasAttribute("async")) notes.push("Blocks the page while it downloads and runs. Add defer, or use type=module.");
      if (script.hasAttribute("defer") && script.hasAttribute("async")) notes.push("Both defer and async are set. Where async is supported, it wins. Choose one.");
      if (!src && script.hasAttribute("defer")) notes.push("defer does nothing on an inline classic script.");
      if (!src && script.hasAttribute("async") && !isModule) notes.push("async does nothing on an inline classic script.");
      if (src && /^http:\/\//i.test(src)) notes.push("Loads over plain http. Use https, or the script can be changed on the way.");
      if (src && script.hasAttribute("integrity") && !script.hasAttribute("crossorigin") && /^https?:\/\//i.test(src)) notes.push("integrity on a script from another site needs crossorigin too.");
      if (src && /^https?:\/\//i.test(src) && !script.hasAttribute("integrity")) notes.push("A script from another site with no integrity check. Make sure you trust it.");
      if (/document\.write/.test(script.textContent)) notes.push("document.write is discouraged. It blocks the parser and can fail in some situations.");
      if (type === "text/javascript" || type === "application/javascript") notes.push("The type attribute is not needed for JavaScript.");
      if (!src && script.textContent.trim() === "") notes.push("Empty script.");
    }
    return {code: script.outerHTML, notes};
  });
  const total = rows.reduce((sum, row) => sum + row.notes.length, 0);
  return (
    <div>
      <label htmlFor="script-source">Paste or edit script tags</label>
      <textarea id="script-source" rows={9} value={source} onChange={(event) => setSource(event.target.value)} style={{width: "100%", fontFamily: "monospace", fontSize: "0.85em"}} />
      <p role="status" style={{margin: "0.4rem 0"}}>Found <strong>{rows.length}</strong> script tag(s) and <strong>{total}</strong> note(s).</p>
      <div style={{maxHeight: "18rem", overflow: "auto"}}>
        <table style={{width: "100%"}}>
          <thead>
            <tr><th scope="col">Tag</th><th scope="col">Notes</th></tr>
          </thead>
          <tbody>
            {rows.map((row, index) => (
              <tr key={index}>
                <td style={{wordBreak: "break-all"}}><code>{row.code}</code></td>
                <td>{row.notes.length === 0 ? "Looks fine." : (<ul style={{margin: 0}}>{row.notes.map((note) => (<li key={note}>{note}</li>))}</ul>)}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <p style={{margin: "0.5rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This reads your text with the browser's real HTML parser. Nothing you type is run or loaded. The notes are guidance, not a full audit.
      </p>
    </div>
  );
}

export function DataBlockDemo() {
  const [text, setText] = useState('{\n  "@context": "https://schema.org",\n  "@type": "Article",\n  "headline": "Sourdough for Beginners",\n  "author": { "@type": "Person", "name": "Ravi" }\n}');
  let parsed = null;
  let error = "";
  try {
    parsed = JSON.parse(text);
  } catch (problem) {
    error = String(problem.message);
  }
  const safe = parsed === null ? "" : JSON.stringify(parsed, null, 2).replace(/</g, "\\u003c");
  const tag = '<script type="application/ld+json">\n' + safe + "\n<\/script>";
  return (
    <div>
      <label htmlFor="data-json">JSON data</label>
      <textarea id="data-json" rows={7} value={text} onChange={(event) => setText(event.target.value)} style={{width: "100%", fontFamily: "monospace", fontSize: "0.85em"}} />
      <p role="status" style={{margin: "0.4rem 0"}}>
        {parsed === null ? "This is not valid JSON: " + error : "Valid JSON. A script block with a data type is never run. Your own code (or a search engine) reads it as text."}
      </p>
      {parsed !== null ? (<pre style={{margin: 0, fontSize: "0.8em"}}><code>{tag}</code></pre>) : null}
      <p style={{margin: "0.5rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        Notice that every less-than sign in the data is written as a code. That stops text such as a closing script tag, inside a value, from ending the block early. Whenever a server builds this block from user data, it must do the same.
      </p>
    </div>
  );
}

HTML builds the **structure** of a page and CSS gives it **style**. But to make a page **do** things (react to clicks, check a form, load fresh data, draw a chart), you need **JavaScript**. And JavaScript enters an HTML page through one element: **`<script>`**.

It is also the **most powerful, and most dangerous,** element you will meet in the head. A script can do anything the page can do. So in this lesson you will learn not only how to add scripts, but how to load them **fast** and **safely**.

## Two Ways to Add a Script

### Inline

```html
<script>
  console.log("This code is written inside the page");
</script>
```

Handy for small snippets. But it is repeated on every page, it cannot be cached by itself, and it is blocked by a strict security policy unless you use a nonce or a hash.

### External

```html
<script src="/js/app.js"></script>
```

The code lives in its own file. It can be **cached** and **shared** between pages, and it works with strict security policies. This is the normal choice for real sites.

:::note A script cannot be both
If a script element has a `src`, the browser **ignores** anything written between its tags. Use one or the other.
:::

## The Problem: Scripts Block the Page

The browser reads your HTML from top to bottom, building the page as it goes. When it meets a plain script, it must **stop**, download the file, run it, and only then carry on. If the script is big or the network is slow, the visitor stares at a blank or half-built page.

That is why you will see old advice to "put scripts at the bottom of the body". It works, but there is a better way: tell the browser how to load the script.

## The Fix: `defer`, `async`, and Modules

| Attribute | Downloads | Runs | Order kept? | Best for |
| :-- | :-- | :-- | :-- | :-- |
| *(none)* | Stops the parser | Immediately | Yes | Rare. Only when a script must run before anything after it |
| **`defer`** | In the background | After the page is read, before `DOMContentLoaded` | **Yes** | Most of your own scripts |
| **`async`** | In the background | As soon as it arrives | **No** | Independent scripts, such as analytics |
| **`type="module"`** | In the background | Like `defer`, by default | Yes | Modern code with `import` and `export` |

A few details:

- **`defer` and `async` only work on external scripts** (with `src`), not inline classic ones. Module scripts are deferred even when they are inline.
- If you set **both** `defer` and `async`, browsers that understand `async` use `async`. Choose one.
- Deferred scripts run **in the order they are written.** Async scripts run in whatever order they finish.
- A module script runs **once,** even if included twice, and always in **strict mode**.

:::tip A simple rule
Use **`defer`** (or `type="module"`) for your own scripts. Use **`async`** only for scripts that depend on nothing and that nothing depends on.
:::

## Where Should the Script Go?

With `defer`, **the head** is the best place. The browser finds the script early, starts downloading at once, and does not block the page.

```html
<head>
  <script src="/js/app.js" defer></script>
</head>
```

A classic script at the **end of the body** also does not block the content above it, but the browser discovers it later, so it starts downloading later. Prefer `defer` in the head.

Inline scripts that need to see an element must come **after** that element, or wait for the page to be ready:

```js
document.addEventListener("DOMContentLoaded", () => {
  // The page is read, so it is safe to find elements here
});
```

### Two Useful Events

| Event | Fires when |
| :-- | :-- |
| `DOMContentLoaded` | The HTML has been read and the page's structure is ready (images may still be loading) |
| `load` | Everything, including images and styles, has finished loading |

## Modules

```html
<script type="module" src="/js/main.js"></script>
```

```js title="main.js"
import { greet } from "./greet.js";
greet("world");
```

Modules let you split code into files with `import` and `export`. They are deferred, run in strict mode, and each has its own scope, so names do not leak into the global space. For old browsers that do not understand modules, `nomodule` marks a fallback, but modern sites rarely need it today.

An **import map** can give short names to module addresses:

```html
<script type="importmap">
  { "imports": { "utils": "/js/utils.js" } }
</script>
```

## Data Blocks: Scripts That Are Not Scripts

If you give `<script>` a `type` that is **not JavaScript**, the browser does not run it. It treats the content as **data**, which your code (or a search engine) can read:

```html
<script type="application/ld+json">
{
  "@context": "https://schema.org",
  "@type": "Article",
  "headline": "Sourdough for Beginners"
}
</script>
```

This is how you add **structured data** (JSON-LD), which can help search engines understand your page. You can also keep your own data in a block like this and read it with JavaScript:

```js
const data = JSON.parse(document.getElementById("page-data").textContent);
```

## `<noscript>`: When JavaScript Is Off

Some visitors have no JavaScript: it may be blocked, it may fail to load, or the device may be too old. The `<noscript>` element shows content only in that case:

```html
<noscript>
  <p>This page works best with JavaScript. The main content is below, but
  the interactive chart is not available.</p>
</noscript>
```

In the **body,** `<noscript>` can hold normal content. In the **head,** it can only hold `<link>`, `<style>`, and `<meta>`.

The better approach is **progressive enhancement:** build the page so the core content and links work **without** JavaScript, then use scripts to add extras.

## Common Script Attributes

| Attribute | Use |
| :-- | :-- |
| `src` | The address of an external script |
| `defer`, `async` | Loading behaviour (see above) |
| `type` | `module`, `importmap`, or a data type. Not needed for ordinary JavaScript |
| `nomodule` | Runs only in browsers that do not understand modules |
| `integrity` | A fingerprint, so the browser can detect a changed file |
| `crossorigin` | How to ask another site for the file. Needed for `integrity` on other sites |
| `referrerpolicy` | How much of your address to send when fetching the script |
| `nonce` | Lets a strict Content-Security-Policy allow this one inline script |
| `fetchpriority` | A hint: `high`, `low`, or `auto` |

## Keeping Scripts Safe

A script has **full power over your page.** It can read what visitors type, change what they see, and send data anywhere. So script safety is the most important security topic in this section.

```html
<script src="https://cdn.example.com/lib@1.2.3/lib.min.js"
        integrity="sha384-…"
        crossorigin="anonymous"
        defer></script>
```

- **Only load scripts you trust,** and pin the **exact version.**
- **Add `integrity` and `crossorigin`,** so a changed file is refused. See [Third-Party Content Security](/tutorials/html/embedded-content/third-party-content-security).
- **Use a Content-Security-Policy** with `script-src`, so only approved scripts can run. This limits the damage of injected code.
- **Never build inline scripts from user input,** and never put user text inside a script block.
- **Avoid `eval`, `document.write`, and inline event attributes** such as `onclick="…"`. They are hard to protect.
- **Use `addEventListener`** instead of inline `onclick` attributes.

## Good Habits

1. **Use `defer` for your own scripts,** in the head.
2. **Use `async` only for independent scripts.**
3. **Use external files** for real code, and inline scripts for tiny snippets.
4. **Use modules** for modern code.
5. **Keep structured data in a `type="application/ld+json"` block.**
6. **Build the page so it works without JavaScript** where possible.
7. **Add `integrity`, `crossorigin`, and a CSP.**
8. **Wait for `DOMContentLoaded`** when you need page elements.
9. **Keep scripts small,** and remove the ones you do not use.

## Accessibility Checklist

- [ ] **Core content and navigation work without JavaScript,** or have a clear fallback.
- [ ] Interactive elements are **real buttons and links,** not `div` elements with click handlers.
- [ ] Everything you can do with a mouse can be done with a **keyboard,** with visible focus.
- [ ] When script changes the page (new content, errors), assistive technology is told, for example with a **live region** or by moving focus.
- [ ] Scripts do not **trap focus** unless it is a proper dialog.
- [ ] Animations respect **`prefers-reduced-motion`**.
- [ ] Time limits and automatic updates can be **paused or extended.**
- [ ] A slow or failed script **does not leave the page unusable.**

## Security and Privacy Notes

- **Every third-party script is a trust decision.** A script from another site can see everything your page shows, read form fields, and send it away. Review who you trust, and why.
- **Cross-site scripting (XSS)** is when an attacker gets their own script into your page. The main defences are **escaping output,** avoiding unsafe methods, and a strict Content-Security-Policy.
- **Use `integrity`** to detect a changed file, but it does not make an untrusted script safe.
- **Analytics and advertising scripts** collect personal data. Laws such as GDPR and similar rules elsewhere may require **consent** before loading them. This is not legal advice, so check the rules for your audience.
- **Do not put secrets in scripts.** Keys, tokens, and passwords in client-side code are public.
- **Escape data in blocks.** A user value containing a closing script tag can break out and run code, so encode `<` when you build JSON blocks on a server.
- **Keep dependencies updated,** and remove unused ones.

## Global Audience Notes

- **Many visitors use slow networks and low-cost phones.** Large scripts can make a site unusable. Ship less code, split it, and load only what a page needs.
- **Parsing and running JavaScript costs more on low-end devices** than downloading it. Test on a slow phone, and not only on your own computer.
- **Do not depend on services that may be blocked or slow** in some regions. Host critical scripts yourself.
- **Make your page work without JavaScript** where you can. It helps on poor connections, and when a script fails to load.
- **Use the browser's own features** (such as native `<dialog>` and `<details>`) in preference to large libraries.
- **Internationalise text and dates in script** with the `Intl` tools, and not with hard-coded English.

## Common Mistakes to Avoid

1. **A plain script in the head,** blocking the page.
2. **`defer` or `async` on an inline classic script.** It does nothing.
3. **Both `defer` and `async` together.**
4. **Using `async` for scripts that depend on each other.**
5. **Writing both `src` and inline code** in one element.
6. **Running code before the elements exist.**
7. **Putting user text inside a script block.**
8. **Third-party scripts without `integrity`,** or with no review.
9. **Inline `onclick` attributes** with a strict policy.
10. **Secrets in script files.**
11. **A page that is blank when JavaScript fails.**
12. **Loading huge libraries** for small jobs.

## Key Takeaways

- **`<script>`** adds JavaScript, **inline** or from an **external file**.
- Plain scripts **block the page.** Use **`defer`** (or **`type="module"`**) for your own code, and **`async`** only for independent scripts.
- Scripts with a **non-JavaScript type** are **data blocks,** such as JSON-LD, and are never run.
- Build pages that work **without JavaScript** where possible, and use **`<noscript>`** for fallbacks.
- A script has **full power** over your page. Use **`integrity`, `crossorigin`, a CSP,** and trusted sources.
- Never put user text inside a script block without encoding it, and never put secrets in client code.
- Keep scripts small. Many visitors use slow networks and low-end devices.
