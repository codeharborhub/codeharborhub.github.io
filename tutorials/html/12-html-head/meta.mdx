import {useState} from 'react';

export const metaSample = [
  '<meta charset="utf-8">',
  '<meta name="viewport" content="width=device-width, initial-scale=1, maximum-scale=1">',
  '<meta name="description" content="Short.">',
  '<meta name="keywords" content="bread, sourdough, baking">',
  '<meta name="robots" content="noindex">',
  '<meta http-equiv="refresh" content="5; url=/new-page">',
  '<meta property="og:title" content="Sourdough for Beginners">',
  '<meta name="theme-color">',
].join("\n");

export function MetaChecker() {
  const [source, setSource] = useState(metaSample);
  const doc = new DOMParser().parseFromString("<!DOCTYPE html><html><head>" + source + "</head><body></body></html>", "text/html");
  const metas = Array.from(doc.head.querySelectorAll("meta"));
  const rows = metas.map((meta) => {
    const name = (meta.getAttribute("name") || "").toLowerCase();
    const equiv = (meta.getAttribute("http-equiv") || "").toLowerCase();
    const property = meta.getAttribute("property") || "";
    const content = meta.getAttribute("content");
    let kind = "unknown";
    let note = "This form is not one this lesson covers.";
    let level = "info";
    if (meta.hasAttribute("charset")) {
      kind = "charset";
      const value = meta.getAttribute("charset").toLowerCase();
      note = value === "utf-8" ? "Good. UTF-8 is the right choice." : "Use utf-8 unless you have a very specific reason.";
      level = value === "utf-8" ? "ok" : "warn";
    } else if (equiv) {
      kind = "http-equiv: " + equiv;
      if (equiv === "refresh") { note = "Automatic refresh or redirect. It can disorient people and break assistive technology. Prefer a server redirect."; level = "warn"; }
      else if (equiv === "content-security-policy") { note = "Works, but a response header is stronger. Some directives (such as frame-ancestors) are ignored here."; level = "info"; }
      else if (equiv === "x-ua-compatible") { note = "Obsolete. Remove it."; level = "warn"; }
      else if (equiv === "content-type") { note = "Legacy form. Use meta charset instead."; level = "warn"; }
      else { note = "Rarely needed. Prefer a real HTTP header if you control the server."; }
    } else if (property) {
      kind = "property: " + property;
      note = property.indexOf("og:") === 0 ? "An Open Graph tag, used by link previews." : "A property-style tag (for example RDFa).";
      level = "ok";
    } else if (name) {
      kind = "name: " + name;
      if (content === null || content === "") { note = "This tag has no content, so it does nothing."; level = "warn"; }
      else if (name === "viewport") {
        const bad = /user-scalable\s*=\s*(no|0)/i.test(content) || /maximum-scale\s*=\s*1(\.0)?(\s|,|$)/i.test(content);
        note = bad ? "This blocks pinch zoom. Remove user-scalable=no and maximum-scale=1." : "Looks fine.";
        level = bad ? "warn" : "ok";
      } else if (name === "description") {
        const length = content.length;
        note = length < 50 ? "Very short (" + length + " characters). Search results work better with a real summary." : length > 160 ? "Long (" + length + " characters). It may be cut off." : "Good length (" + length + " characters).";
        level = length < 50 || length > 160 ? "warn" : "ok";
      } else if (name === "robots") {
        const hidden = /noindex|none/i.test(content);
        note = hidden ? "This asks search engines NOT to list the page. Make sure that is what you want." : "Fine.";
        level = hidden ? "warn" : "ok";
      } else if (name === "keywords") { note = "Major search engines ignore it. It does no harm, but you can skip it."; }
      else if (name === "theme-color") { note = "Suggests a toolbar colour in some browsers."; level = "ok"; }
      else if (name === "color-scheme") { note = "Tells the browser which colour schemes the page supports."; level = "ok"; }
      else { note = "A named tag. Make sure something actually reads it."; }
    }
    return {code: meta.outerHTML, kind, note, level};
  });
  const descriptions = rows.filter((row) => row.kind === "name: description").length;
  const missing = [];
  if (!doc.head.querySelector("meta[charset]")) missing.push("meta charset");
  if (!doc.head.querySelector('meta[name="viewport"]')) missing.push("meta viewport");
  if (!doc.head.querySelector('meta[name="description"]')) missing.push("meta description");
  const marker = (level) => (level === "ok" ? "OK" : level === "warn" ? "Check" : "Note");
  return (
    <div>
      <label htmlFor="meta-source">Paste or edit meta tags</label>
      <textarea id="meta-source" rows={9} value={source} onChange={(event) => setSource(event.target.value)} style={{width: "100%", fontFamily: "monospace", fontSize: "0.85em"}} />
      <div role="status" style={{marginTop: "0.4rem"}}>
        <p style={{margin: "0 0 0.3rem"}}>Found <strong>{rows.length}</strong> meta tag(s).{missing.length > 0 ? " Not found: " + missing.join(", ") + "." : " Charset, viewport and description are all present."}{descriptions > 1 ? " There is more than one description, and only one should be used." : ""}</p>
      </div>
      <div style={{maxHeight: "18rem", overflow: "auto"}}>
        <table style={{width: "100%"}}>
          <thead>
            <tr><th scope="col">Tag</th><th scope="col">Kind</th><th scope="col">Result</th><th scope="col">What it means</th></tr>
          </thead>
          <tbody>
            {rows.map((row, index) => (
              <tr key={index}>
                <td style={{wordBreak: "break-all"}}><code>{row.code}</code></td>
                <td>{row.kind}</td>
                <td><strong>{marker(row.level)}</strong></td>
                <td>{row.note}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <p style={{margin: "0.5rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This reads your text with your browser's real HTML parser. Nothing is run, and nothing is sent anywhere. The notes are guidance, not a complete audit.
      </p>
    </div>
  );
}

A web page is like a **book**. The cover shows the title, but the **back cover** and the **library catalogue card** hold other facts: a short summary, the language, the author, the category, who may borrow it. Visitors rarely see those, but librarians, bookshops, and search systems depend on them.

In HTML, the **`<meta>` element** is that catalogue card. It stores facts **about** the page that no other element covers. In this lesson you will meet its different forms, the most useful tags, and a few that are best avoided.

## The Anatomy of a Meta Tag

`<meta>` is a **void element**: it has no content and no closing tag. Everything is in its attributes. It comes in a few forms:

| Form | Example | Used for |
| :-- | :-- | :-- |
| **charset** | `<meta charset="utf-8">` | How to read the file |
| **name and content** | `<meta name="description" content="…">` | Named facts about the page |
| **http-equiv and content** | `<meta http-equiv="refresh" content="…">` | A few instructions that act like HTTP headers |
| **property and content** | `<meta property="og:title" content="…">` | Link previews (Open Graph) and similar |

A tag in the **name** form needs both attributes. Without `content`, it does nothing.

:::note Other forms exist
There are also `itemprop` (for microdata) and `media` (used with `theme-color`). You will rarely need them at this stage.
:::

## Charset and Viewport

Two meta tags belong on **every** page. They have their own lessons, so here is a quick summary:

```html
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
```

- **Charset** lets the browser read your letters correctly. See [Charset](./charset).
- **Viewport** makes the page fit phones. See [Viewport](./viewport).

## The Description

```html
<meta name="description" content="A gentle, step-by-step guide to baking your first sourdough loaf.">
```

A short summary of the page. Search engines **may** show it under your title. They often choose text from the page instead, if it fits the search better, so treat it as a strong suggestion, not a command.

How to write a good one:

- **One or two sentences,** around 120 to 160 characters.
- **Unique for each page.** Copied descriptions give no information.
- **Written for people:** say what the visitor will get.
- **Honest.** Do not promise what the page does not deliver.
- **No keyword lists.**

| Weak | Better |
| :-- | :-- |
| `Bread.` | `A gentle, step-by-step guide to baking your first sourdough loaf, with no special equipment.` |
| `bread, bake, sourdough, loaf, cheap, best` | `Learn how to feed a starter, shape the dough, and bake a crusty loaf at home.` |

## Telling Search Engines What to Do: robots

```html
<meta name="robots" content="index, follow">
```

| Value | Meaning |
| :-- | :-- |
| `index` | The page may be listed (the default) |
| `noindex` | Please do **not** list this page |
| `follow` | Links on the page may be followed (the default) |
| `nofollow` | Do not follow links on this page |
| `noindex, nofollow` | Neither list the page nor follow its links |

Use `noindex` for pages that should not appear in search results: thank-you pages, internal search results, or drafts.

:::danger Two big warnings about robots
1. **`noindex` is a request, not security.** Well-behaved search engines respect it, but anyone with the link can still open the page. Protect private pages with a login.
2. **A single `noindex` left on by accident can remove a whole site from search results.** It is a common launch-day mistake. Check your live pages.
:::

## Colour Hints

### theme-color

```html
<meta name="theme-color" content="#1d4ed8">
<meta name="theme-color" content="#0b1220" media="(prefers-color-scheme: dark)">
```

Some mobile browsers use this colour for their toolbar, so the page feels more like an app. Use the `media` attribute to supply different colours for light and dark. Browser support varies, and it is only a hint.

### color-scheme

```html
<meta name="color-scheme" content="light dark">
```

This tells the browser your page supports **both** light and dark schemes, so it can draw things like the default page background, form controls, and scrollbars in the right colours **before** your CSS loads. It prevents a white flash on dark pages.

## http-equiv: Header-Like Instructions

`http-equiv` means "equivalent to an HTTP header". It lets a page state a few instructions that a server would normally send:

```html
<meta http-equiv="content-security-policy" content="default-src 'self'">
```

| Value | What it does | Advice |
| :-- | :-- | :-- |
| `content-security-policy` | Restricts what the page may load | Works, but a real header is stronger. Some directives do not work in a meta tag |
| `refresh` | Reloads or redirects after some seconds | **Avoid.** See below |
| `content-type` | Old way to declare the character set | Use `<meta charset>` instead |
| `x-ua-compatible` | Old Internet Explorer setting | Obsolete. Remove it |

### Why You Should Avoid `refresh`

```html
<meta http-equiv="refresh" content="5; url=/new-page">
```

An automatic redirect or reload that a visitor did not ask for causes real problems:

- A screen reader user may be **moved away before they have finished reading.**
- People who need **more time** to read, or who use switch devices, can be locked out.
- It breaks the browser's back button in many cases.

Use a **server redirect** (an HTTP 301 or 302) for moved pages. If you must refresh, give a **link** and let the visitor choose. This relates to the accessibility guidance on giving users control over time limits and unexpected changes.

## Open Graph: Link Previews

When you paste a link in a chat or a social app, you often see a **card** with a picture, title, and summary. Many of those apps read **Open Graph** tags:

```html
<meta property="og:type" content="article">
<meta property="og:title" content="Sourdough for Beginners">
<meta property="og:description" content="A gentle guide to your first loaf.">
<meta property="og:url" content="https://example.com/sourdough">
<meta property="og:image" content="https://example.com/images/sourdough-1200x630.jpg">
<meta property="og:image:alt" content="A golden loaf on a wooden board">
<meta name="twitter:card" content="summary_large_image">
```

Notes:

- Use **full `https://` addresses**, not relative ones.
- A **wide image**, roughly 1200 by 630 pixels, is a common recommendation for large cards. Check the platform's current guidance.
- Always add **alt text** for the image.
- Apps **cache** previews. After you change tags, you may need the platform's debugging tool to refresh.
- Not every app uses every tag. Some fall back to the `<title>` and `description`.

Try it: change the fields, and see the card and the tags it produces.

## Other Meta Tags You May See

| Tag | Status |
| :-- | :-- |
| `<meta name="author" content="…">` | Allowed, and harmless. Gives a name, which is public |
| `<meta name="generator" content="…">` | Records which tool made the page. Reveals your software version |
| `<meta name="referrer" content="strict-origin-when-cross-origin">` | Controls how much of your address is sent to other sites |
| `<meta name="keywords" content="…">` | **Ignored** by major search engines. Skip it |
| `<meta name="format-detection" content="telephone=no">` | Stops some mobile browsers turning numbers into phone links |
| `<meta name="google-site-verification" content="…">` | Proves site ownership to a service. Use the one your service gives you |

## Good Habits

1. **Always include charset and viewport.**
2. **Write a unique description for each page.**
3. **Use `robots` carefully,** and check live pages for accidental `noindex`.
4. **Do not use `refresh`.** Use a server redirect.
5. **Add `color-scheme`** if your page supports dark mode.
6. **Include Open Graph tags** on pages people share.
7. **Do not stuff keywords.**
8. **Remove obsolete tags** to keep the head light.

## Accessibility Checklist

- [ ] The viewport tag **does not** disable zoom.
- [ ] There is **no** automatic `refresh` that moves people without warning.
- [ ] `og:image:alt` (or equivalent) is provided for previews.
- [ ] `color-scheme` matches the schemes your CSS really supports.
- [ ] `theme-color` keeps good contrast with the browser's interface.
- [ ] Descriptions are written in plain language, in the page's language.

## Security and Privacy Notes

- **Meta tags are public.** Never put secrets, tokens, or private notes in them.
- **`generator` and `author`** can reveal your software, its version, and names. Remove them if that is a concern.
- **`noindex` does not protect anything.** It only asks search engines to stay away. Use authentication for private pages.
- **A Content-Security-Policy in a meta tag** is weaker than the header form: it only applies after the browser has read the tag, and it cannot set some directives.
- **`referrer`** helps keep private paths and search terms from being sent to other sites. Pick a strict value if your addresses contain sensitive data.
- **Escape dynamic content.** If a meta `content` comes from a database or a visitor, escape quotes and angle brackets, or it can break out of the attribute.
- **Open Graph tags reveal what you share.** Do not put private details in descriptions that will be shown to anyone who sees a link.

## Global Audience Notes

- **Write the description and Open Graph text in the page's language,** and match `lang` and `og:locale` (for example `og:locale` set to `hi_IN`).
- **Make a separate set of tags for each language version,** and connect them with `hreflang` links.
- **Descriptions in languages with wider characters** (such as Chinese, Japanese, and Korean) are shorter in characters but not in width. Test them.
- **Use `color-scheme` and `theme-color`** to respect people's preferences in every region.
- **Preview images with text** should be readable in the language of the page, and have alt text.
- **Do not rely on country-specific services** for verification tags, if your audience is global.

## Common Mistakes to Avoid

1. **No description,** or the same one on every page.
2. **A keyword list** as the description.
3. **An accidental `noindex`,** removing the page from search.
4. **Using `refresh`** for redirects.
5. **Blocking zoom** in the viewport tag.
6. **A `name` tag without `content`.**
7. **Relative addresses in Open Graph tags.**
8. **A huge or tiny share image.**
9. **Two descriptions on one page.**
10. **Relying on `keywords`,** which search engines ignore.
11. **Secrets in meta tags.**
12. **Forgetting that apps cache previews.**

## Key Takeaways

- `<meta>` stores facts about the page in its attributes. It has **no closing tag**.
- Every page needs **charset** and **viewport**, and a unique **description** helps.
- **`robots`** controls listing, but `noindex` is a request, not security.
- **`theme-color`** and **`color-scheme`** improve how the browser's interface and defaults look.
- Avoid **`refresh`**, obsolete tags, and keyword lists.
- **Open Graph** tags make link previews work. Use full addresses, a good image, and alt text.
- Meta tags are public, so keep secrets out and escape dynamic values.
