export function esc(value) {
  return String(value).replace(/&/g, "&amp;").replace(/"/g, "&quot;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

export function validHandle(value) {
  return /^@[A-Za-z0-9_]{1,15}$/.test(value.trim());
}

export function FallbackLab() {
  const [has, setHas] = useState({card: true, ttitle: false, otitle: true, tdesc: false, odesc: true, mdesc: true, timage: false, oimage: true});
  const toggle = (key) => setHas({...has, [key]: !has[key]});
  const pick = (list) => {
    const found = list.find((item) => has[item.key]);
    return found ? found.label : null;
  };
  const title = pick([{key: "ttitle", label: "twitter:title"}, {key: "otitle", label: "og:title"}, {key: "ptitle", label: "the page title"}]) || "the page title (always present)";
  const description = pick([{key: "tdesc", label: "twitter:description"}, {key: "odesc", label: "og:description"}, {key: "mdesc", label: "the meta description"}]);
  const image = pick([{key: "timage", label: "twitter:image"}, {key: "oimage", label: "og:image"}]);
  const items = [
    {key: "card", label: "twitter:card"},
    {key: "ttitle", label: "twitter:title"},
    {key: "otitle", label: "og:title"},
    {key: "tdesc", label: "twitter:description"},
    {key: "odesc", label: "og:description"},
    {key: "mdesc", label: "meta name=\"description\""},
    {key: "timage", label: "twitter:image"},
    {key: "oimage", label: "og:image"},
  ];
  return (
    <div>
      <fieldset style={{margin: "0 0 0.7rem"}}>
        <legend>Which tags does your page have?</legend>
        {items.map((item) => (
          <label key={item.key} style={{display: "block"}}><input type="checkbox" checked={has[item.key]} onChange={() => toggle(item.key)} /> <code>{item.label}</code></label>
        ))}
      </fieldset>
      <p style={{margin: "0 0 0.3rem"}}><strong>What a card is likely to use:</strong></p>
      <ul style={{margin: "0 0 0.6rem", paddingLeft: "1.2rem"}} aria-live="polite">
        <li><strong>Card:</strong> {has.card ? "A rich card (the type named in twitter:card)." : "No twitter:card, so X may show only a plain link, even if Open Graph tags exist."}</li>
        <li><strong>Title:</strong> {title}</li>
        <li><strong>Description:</strong> {description || "nothing, so the card may show no description"}</li>
        <li><strong>Image:</strong> {image || "nothing, so the card is text only"}</li>
      </ul>
      <p style={{margin: "0 0 0.3rem"}}>
        {has.card && !has.timage && has.oimage && !has.ttitle && !has.tdesc ? "This is the lean, popular setup: your Open Graph tags do the work, and one extra tag turns on the card." : null}
        {!has.card && (has.otitle || has.oimage) ? "Tip: add twitter:card, and your Open Graph tags will be used for the rest." : null}
      </p>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This shows the commonly documented order of fallbacks, from the most specific tag to the least. Platforms can change their behaviour at any time, so check the current documentation and test your page.
      </p>
    </div>
  );
}

export function TwitterCheckerInner() {
  const [which, setWhich] = useState("messy");
  const [html, setHtml] = useState(twSamples.messy);
  const doc = new DOMParser().parseFromString("<!doctype html>" + html, "text/html");
  const issues = [];
  const notes = [];
  const metaByName = {};
  Array.from(doc.querySelectorAll("meta[name], meta[property]")).forEach((el) => {
    const name = (el.getAttribute("name") || el.getAttribute("property") || "").trim();
    (metaByName[name] = metaByName[name] || []).push({value: (el.getAttribute("content") || "").trim(), usedProperty: !el.hasAttribute("name")});
  });
  const first = (key) => (metaByName[key] ? metaByName[key][0].value : "");
  Object.keys(metaByName).filter((key) => key.indexOf("twitter:") === 0).forEach((key) => {
    if (metaByName[key].length > 1) issues.push(key + " appears " + metaByName[key].length + " times. Use it once.");
    if (metaByName[key].some((item) => item.usedProperty)) notes.push(key + " uses property. The convention for Twitter tags is name, which is the safest choice.");
  });
  const card = first("twitter:card");
  if (!metaByName["twitter:card"]) issues.push("No twitter:card. Without it, X may show only a plain link.");
  else if (["summary", "summary_large_image", "app", "player"].indexOf(card) === -1) issues.push("twitter:card \"" + card + "\" is not a known value. Use summary or summary_large_image.");
  else if (card === "app" || card === "player") notes.push("The \"" + card + "\" card type is specialised and has limited support today. Most pages should use summary or summary_large_image.");
  if (metaByName["twitter:player"]) notes.push("twitter:player belongs to the player card, which has limited support. Most pages do not need it.");
  const title = first("twitter:title") || first("og:title") || (doc.querySelector("title") ? doc.querySelector("title").textContent.trim() : "");
  const titleFrom = first("twitter:title") ? "twitter:title" : first("og:title") ? "og:title" : "the page title";
  if (title === "") issues.push("No title found in twitter:title, og:title, or the page title.");
  else if (title.length > 70) notes.push("The title (from " + titleFrom + ") is " + title.length + " characters, so it may be cut off.");
  else if (/^(home|untitled|rye bread)$/i.test(title)) notes.push("The title \"" + title + "\" (from " + titleFrom + ") is very short or generic. Say what the page is about.");
  const descFrom = first("twitter:description") ? "twitter:description" : first("og:description") ? "og:description" : first("description") ? "the meta description" : "";
  if (!descFrom) notes.push("No description in twitter:description, og:description, or the meta description. The card may show none.");
  const image = first("twitter:image") || first("og:image");
  const imageFrom = first("twitter:image") ? "twitter:image" : "og:image";
  if (!image) issues.push("No image in twitter:image or og:image. The card will be text only.");
  else {
    if (!/^https:\/\//i.test(image)) issues.push(imageFrom + " \"" + image + "\" must be a full https address.");
    else if (/\.svg(\?|$)/i.test(image)) issues.push(imageFrom + " is an SVG. X does not accept SVG images. Use JPG, PNG, WebP, or GIF.");
    if (!first("twitter:image:alt") && !first("og:image:alt")) notes.push("No image description. Add twitter:image:alt.");
  }
  ["twitter:site", "twitter:creator"].forEach((key) => {
    const value = first(key);
    if (value && !/^@[A-Za-z0-9_]{1,15}$/.test(value)) issues.push(key + " \"" + value + "\" should be a handle like @thebreadbin: an @ sign, then up to 15 letters, numbers, or underscores.");
  });
  if (!metaByName["twitter:site"]) notes.push("No twitter:site. It is optional, but it connects the card to your account.");
  if (card === "summary_large_image" && !image) notes.push("A large image card needs an image.");
  const choose = (value) => { setWhich(value); setHtml(twSamples[value]); };
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>Which page do you want to check?</legend>
        <label style={{marginRight: "1rem"}}><input type="radio" name="twc-sample" checked={which === "messy"} onChange={() => choose("messy")} /> A messy page</label>
        <label><input type="radio" name="twc-sample" checked={which === "fixed"} onChange={() => choose("fixed")} /> The fixed version</label>
      </fieldset>
      <p style={{margin: "0 0 0.5rem"}}>
        <label htmlFor="twc-html">HTML head (edit it freely)</label><br />
        <textarea id="twc-html" rows={11} value={html} onChange={(event) => setHtml(event.target.value)} style={{width: "100%", fontFamily: "monospace"}} />
      </p>
      <p style={{margin: "0 0 0.3rem"}}><strong>What a card would use:</strong> title from {titleFrom}, description from {descFrom || "nothing"}, image from {image ? imageFrom : "nothing"}.</p>
      <p role="status" style={{margin: "0 0 0.4rem"}}><strong>{issues.length}</strong> problems, <strong>{notes.length}</strong> suggestions.</p>
      {issues.length === 0 && notes.length === 0 ? <p style={{margin: 0}}>Nothing to report.</p> : null}
      <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
        {issues.map((issue, index) => (<li key={"i" + index}><strong>Problem:</strong> {issue}</li>))}
        {notes.map((note, index) => (<li key={"n" + index}><strong>Suggestion:</strong> {note}</li>))}
      </ul>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This reads the HTML only. It cannot fetch your image, check robots.txt, or see what X has cached. Test the live page by drafting a post with the link.
      </p>
    </div>
  );
}

## What Twitter Cards Are

A **Twitter Card** is the preview that appears under a link in a post: a picture, a title, and a line of description, instead of just a bare address. The card is built from `<meta>` tags in your page's head, read by X's crawler (called **Twitterbot**) when the link is shared.

- Like Open Graph, cards do **not** directly affect your search ranking. They make shared links more inviting.
- Unlike Open Graph, the conventional attribute is **`name`,** not `property`: `<meta name="twitter:card" ...>`.
- Other apps do not usually read them. Chat apps, Slack, Discord, WhatsApp, and Mastodon mostly use **Open Graph.** That is one more reason to write good Open Graph tags first.

## The Card Types

| Card | What it looks like | Best for |
| :-- | :-- | :-- |
| `summary` | A small square picture beside the title and description | Pages without a strong picture, or with a logo |
| `summary_large_image` | A large picture above the title and description | Articles, recipes, products, and anything with a good image |

Two other types (`app` and `player`) exist for specialised uses, and have limited support today. Most pages only ever need one of the two above. When in doubt, use **`summary_large_image`.**

## The Tags

| Tag | What it does |
| :-- | :-- |
| `twitter:card` | **The one that turns the card on.** Its value is the card type |
| `twitter:title` | The headline of the card |
| `twitter:description` | A line or two of text |
| `twitter:image` | The picture |
| `twitter:image:alt` | A text description of the picture |
| `twitter:site` | The account for the website (`@thebreadbin`) |
| `twitter:creator` | The account of the author (`@ravipatel`) |

Handles are written with an **@ sign,** and X handles are letters, numbers, and underscores, up to 15 characters.

## Fallbacks: Write Less

If a `twitter:` tag is missing, X usually falls back to the matching Open Graph tag, and then to the page itself:

| Needed | First choice | Then | Then |
| :-- | :-- | :-- | :-- |
| Title | `twitter:title` | `og:title` | the page `<title>` |
| Description | `twitter:description` | `og:description` | the meta description |
| Image | `twitter:image` | `og:image` | nothing |

The one thing that does **not** have a fallback is **`twitter:card`.** Without it, you may get only a plain link, even if your Open Graph tags are perfect.

So the **lean setup** is:

1. Write good **Open Graph** tags.
2. Add **`twitter:card`.**
3. Optionally add `twitter:site` and `twitter:image:alt`.

Add the full set of `twitter:` tags only when you want the card to say something **different** from your Open Graph tags.

## Images

Per X's public guidance (check the current documentation, as sizes change):

| Card | Shape | Suggested size | Notes |
| :-- | :-- | :-- | :-- |
| `summary_large_image` | About 2 to 1 | 1200 by 600 is a good target. At least 300 by 157 | Larger images get cropped to fit |
| `summary` | Square, 1 to 1 | At least 144 by 144. 600 by 600 is a good target | Shown as a small thumbnail |

- **Formats:** JPG, PNG, WebP, and GIF (usually the first frame). Not SVG.
- **File size:** keep it well under a few megabytes. Smaller is faster.
- **Full https address,** public, with no login.
- **Keep key content in the middle,** since edges may be cropped.
- **Use an image of about 2 to 1 for both Open Graph and Twitter** if you want one file to serve both. A 1200 by 630 image (Open Graph's usual size) is close enough to 2 to 1 for a large card, and many sites use it for both.
- **Add `twitter:image:alt`** so the image has a description. X limits alt text length, so keep it short and plain.

## Do Not Block Twitterbot

X's crawler obeys your **robots.txt.** If your file blocks it, or blocks the folder where your images live, cards will fail to appear.

```text title="robots.txt"
# This would break cards
User-agent: Twitterbot
Disallow: /
```

Check that Twitterbot is allowed on your pages **and** on your image address. See [Robots Meta](./robots-meta). Also remember that pages and images behind a login cannot be fetched, so cards for them show nothing.

## Caching and Testing

- **X caches cards,** like other platforms. After you change a tag or image, an old card may linger for a while. Using a new image address (such as a version number in the file name) is the most reliable fix.
- **X has retired its old Card Validator tool,** so do not rely on it. Check X's current help pages in case a new one exists. The most reliable test is to **draft a post** with your link in it, and look at the preview before you publish. Check X's current help pages for any new tool.
- **Test the Open Graph tags too,** with the Facebook Sharing Debugger and the LinkedIn Post Inspector. If your Open Graph is right, the fallback works.

## Security and Privacy Notes

- **Escape every value.** If a title or description comes from a database or a user, a stray quote or angle bracket can break your tags and inject content. Escape before you write it into `content="..."`. The builder above does this.
- **Handles identify real people.** `twitter:creator` links a page to a person's account. Only use a handle with the person's consent, and prefer your organisation's account over a personal one.
- **Private pages leak through cards.** If someone shares a link to a private page, X may fetch it and show its title and image to everyone. Put generic text in the tags of logged-in pages, and do not expose private details.
- **Pictures can reveal more than you expect.** Check images for names, faces, and locations, and remove hidden location data (EXIF) from photos.
- **Do not mislead.** A card that says one thing while the page says another may be treated as spam or a scam by the platform.
- **Do not put tracking codes in the image address** that identify individual visitors. The card is shown to many people.
- **Watch for tampering.** If a hacked page has its tags changed, shared links can show spam. Check your live tags from time to time.
- **Crawlers appear in your logs.** Twitterbot fetching your page is normal. If you restrict access by address or user agent, allow it deliberately.

## Global Audience Notes

- **Write each language version's tags in that language,** and give it its own image if the picture contains text.
- **Check where your readers share.** X is not equally popular everywhere, and other apps (which read Open Graph) may matter more. Do not neglect Open Graph.
- **Text direction:** cards for right-to-left languages such as Arabic and Hebrew should be tested in the real preview.
- **Titles and descriptions are cut at different lengths** in different scripts. Preview each language.
- **Handles are not translated.** Use the account that matches the audience, such as a separate account per language, if you have one.
- **Slow networks:** a large image may be slow to load or fail on a poor connection. Keep files small.
- **Access and rules vary by country.** The platform may be limited or unavailable in some places. Do not depend on it as your only channel.

## Common Mistakes to Avoid

1. **No `twitter:card` tag,** so no card appears.
2. **Relying on Open Graph alone,** and forgetting the card tag.
3. **A relative or SVG image.**
4. **An image of the wrong shape,** which gets cropped badly.
5. **Handles without the `@`,** or with spaces.
6. **Blocking Twitterbot** in robots.txt.
7. **A private or login-protected image.**
8. **Using `property` instead of `name`.**
9. **Duplicate tags that disagree.**
10. **A stale card that is never refreshed.**
11. **No alt text** for the image.
12. **Never previewing** the card before posting.

- **Twitter Cards control how your link looks on X.** The tags still use the `twitter:` prefix.

- **Open Graph is the fallback** for title, description, and image, so write it first.
- **Use `summary_large_image`** for most pages, with a wide image of about 2 to 1.
