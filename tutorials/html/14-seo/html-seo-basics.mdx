export function checkUrl(input) {
  const raw = input.trim();
  const results = [];
  if (raw === "") return results;
  let path = raw;
  let query = "";
  const q = raw.indexOf("?");
  if (q !== -1) { path = raw.slice(0, q); query = raw.slice(q + 1); }
  const noOrigin = path.replace(/^https?:\/\/[^/]+/i, "");
  results.push({ok: !/^http:\/\//i.test(raw), text: /^http:\/\//i.test(raw) ? "Uses http. Prefer https." : "Uses https, or a relative path."});
  results.push({ok: noOrigin === noOrigin.toLowerCase(), text: noOrigin === noOrigin.toLowerCase() ? "All lowercase." : "Has capital letters. Paths can be case sensitive, so lowercase is safer."});
  results.push({ok: !/\s|%20/.test(raw), text: /\s|%20/.test(raw) ? "Contains spaces. Use hyphens." : "No spaces."});
  results.push({ok: noOrigin.indexOf("_") === -1, text: noOrigin.indexOf("_") === -1 ? "No underscores." : "Has underscores. Hyphens are easier to read as word breaks."});
  const params = query === "" ? 0 : query.split("&").length;
  results.push({ok: params <= 1, text: params === 0 ? "No query parameters." : params + " query parameter" + (params > 1 ? "s" : "") + (params > 1 ? ". Many parameters can create endless duplicate pages." : ".")});
  results.push({ok: noOrigin.length <= 75, text: noOrigin.length <= 75 ? "A reasonable length (" + noOrigin.length + " characters)." : "Long (" + noOrigin.length + " characters). Shorter addresses are easier to read and share."});
  const lastPart = noOrigin.split("/").filter(Boolean).pop() || "";
  const hasWords = /[a-z]{4,}/i.test(lastPart);
  results.push({ok: hasWords, text: hasWords ? "Has readable words." : "No readable words. An address like /recipes/rye-bread tells people what to expect."});
  results.push({ok: !/(sessionid|sid|phpsessid)=/i.test(query), text: /(sessionid|sid|phpsessid)=/i.test(query) ? "Contains a session id. This is private data and creates duplicate pages." : "No session id."});
  return results;
}

export function BasicsCheckerInner() {
  const [which, setWhich] = useState("messy");
  const [html, setHtml] = useState(basicsSamples.messy);
  const doc = new DOMParser().parseFromString(html, "text/html");
  const results = [];
  const check = (ok, good, bad, level) => results.push({ok, text: ok ? good : bad, level: level || "problem"});
  check(/^\s*<!doctype html/i.test(html), "Starts with the HTML doctype.", "No doctype. Start with <!doctype html> so browsers use standards mode.");
  const lang = doc.documentElement.getAttribute("lang");
  check(!!lang, "The html element has lang=\"" + lang + "\".", "No lang attribute on the html element.");
  check(!!doc.querySelector("meta[charset]"), "Character encoding is declared.", "No meta charset. Add <meta charset=\"utf-8\">.");
  check(!!doc.querySelector('meta[name="viewport"]'), "The viewport meta tag is present.", "No viewport meta tag, so mobile pages may not scale properly.");
  const title = doc.querySelector("title");
  const titleText = title ? title.textContent.trim() : "";
  if (!title || titleText === "") check(false, "", "No title, or an empty one.");
  else if (/^(home|untitled|document|index|page)$/i.test(titleText)) check(false, "", "The title \"" + titleText + "\" is too generic. Say what the page is about.");
  else if (titleText.length > 65) check(false, "", "The title is " + titleText.length + " characters, so it may be cut off.", "note");
  else check(true, "The title is present and specific (" + titleText.length + " characters).", "");
  const desc = doc.querySelector('meta[name="description"]');
  const descText = desc ? (desc.getAttribute("content") || "").trim() : "";
  check(descText !== "", "A meta description is present (" + descText.length + " characters).", "No meta description. Search engines may pick a snippet from the page.", "note");
  const h1s = doc.querySelectorAll("h1");
  check(h1s.length === 1, "Exactly one h1.", h1s.length === 0 ? "No h1. Add one clear main heading." : h1s.length + " h1 headings. Most pages should have one.");
  check(!!doc.querySelector("main"), "There is a main element.", "No main element. Wrap the main content in one.", "note");
  check(!!doc.querySelector('link[rel="canonical"]'), "A canonical link is present.", "No canonical link. Add one if the page can be reached at more than one address.", "note");
  const robots = doc.querySelector('meta[name="robots"]');
  if (robots && /noindex/i.test(robots.getAttribute("content") || "")) check(false, "", "The robots meta says noindex. This page asks not to be listed. Is that on purpose?");
  if (doc.querySelector('meta[name="keywords"]')) check(false, "", "A meta keywords tag. Major search engines ignore it, so you can remove it.", "note");
  const noHref = doc.querySelectorAll("a:not([href])");
  check(noHref.length === 0, "Every a element has an href.", noHref.length + " a element" + (noHref.length > 1 ? "s have" : " has") + " no href. Crawlers cannot follow it.");
  const vague = Array.from(doc.querySelectorAll("a[href]")).filter((el) => /^(click here|here|read more|more|link)$/i.test((el.textContent || "").trim()));
  check(vague.length === 0, "Link texts are descriptive.", vague.length + " link" + (vague.length > 1 ? "s say" : " says") + " something vague, like \"" + (vague[0] ? vague[0].textContent.trim() : "") + "\". Say where it goes.");
  const imgs = Array.from(doc.querySelectorAll("img"));
  const noAlt = imgs.filter((el) => !el.hasAttribute("alt"));
  const badAlt = imgs.filter((el) => /^(image|photo|picture|img|graphic)?\s*\d*$/i.test((el.getAttribute("alt") || "").trim()) && el.hasAttribute("alt") && (el.getAttribute("alt") || "").trim() !== "");
  const badName = imgs.filter((el) => /^(img|dsc|image|photo)[_-]?\d+/i.test((el.getAttribute("src") || "").split("/").pop()));
  if (imgs.length > 0) {
    check(noAlt.length === 0, "Every image has an alt attribute.", noAlt.length + " image" + (noAlt.length > 1 ? "s have" : " has") + " no alt attribute.");
    check(badAlt.length === 0, "Alt text is descriptive.", badAlt.length + " image" + (badAlt.length > 1 ? "s have" : " has") + " vague alt text such as \"image\". Describe what the image shows.");
    check(badName.length === 0, "Image file names are meaningful.", badName.length + " image file name" + (badName.length > 1 ? "s look" : " looks") + " like a camera default (such as IMG_0042). A clear name helps a little.", "note");
    const noSize = imgs.filter((el) => !el.hasAttribute("width") || !el.hasAttribute("height"));
    check(noSize.length === 0, "Images have width and height.", noSize.length + " image" + (noSize.length > 1 ? "s have" : " has") + " no width and height. The page may jump around as images load.", "note");
  }
  const problems = results.filter((item) => !item.ok && item.level === "problem").length;
  const notes = results.filter((item) => !item.ok && item.level === "note").length;
  const choose = (value) => { setWhich(value); setHtml(basicsSamples[value]); };
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>Which page do you want to check?</legend>
        <label style={{marginRight: "1rem"}}><input type="radio" name="bs-sample" checked={which === "messy"} onChange={() => choose("messy")} /> A messy page</label>
        <label><input type="radio" name="bs-sample" checked={which === "fixed"} onChange={() => choose("fixed")} /> The fixed version</label>
      </fieldset>
      <p style={{margin: "0 0 0.5rem"}}>
        <label htmlFor="bs-html">Full HTML page (edit it freely)</label><br />
        <textarea id="bs-html" rows={14} value={html} onChange={(event) => setHtml(event.target.value)} style={{width: "100%", fontFamily: "monospace"}} />
      </p>
      <p role="status" style={{margin: "0 0 0.4rem"}}><strong>{problems}</strong> problems, <strong>{notes}</strong> suggestions.</p>
      <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
        {results.map((item, index) => (
          <li key={index} style={{marginBottom: "0.25rem"}}><strong>{item.ok ? "OK:" : (item.level === "note" ? "Suggestion:" : "Problem:")}</strong> {item.text}</li>
        ))}
      </ul>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This is a small teaching checker. It reads the HTML only, so it cannot judge content quality, speed, links from other sites, or what a search engine will really do. Use it with Search Console and Lighthouse, not instead of them.
      </p>
    </div>
  );
}

## What a Search Engine Receives

When a crawler visits your page, it first downloads the **raw HTML** from your server. It looks for:

- The **head:** title, description, canonical, robots, language.
- The **text** in the body, and how it is organised into headings and sections.
- The **links,** to find more pages.
- The **images,** with their alt text.
- Any **structured data.**

Many search engines can then also **run your JavaScript** to see the final page, but that step can be slower, can be limited, and does not happen with every crawler. So the safest rule is: **put the important things in the HTML itself.**

:::tip See what a crawler sees
Right-click your page and choose **View page source.** This shows the raw HTML the server sent, before any scripts ran. If your key text, headings, and links are missing there, they may be missing for crawlers too. (The browser's Elements panel in developer tools shows the page after scripts have run, which is different.)
:::

## The Head Essentials

| Tag | Why it matters |
| :-- | :-- |
| `<!doctype html>` | Tells browsers to use standards mode |
| `<html lang="en">` | Says what language the page is in |
| `<meta charset="utf-8">` | Makes sure characters display correctly |
| `<meta name="viewport" ...>` | Makes the page scale properly on phones |
| `<title>` | Names the page in results and tabs |
| `<meta name="description">` | Offers a summary for the result snippet |
| `<link rel="canonical">` | Names the main address of the page |

You can read the basics of the first few in [Title](/tutorials/html/html-head/title), [Viewport](/tutorials/html/html-head/viewport), and [Charset](/tutorials/html/html-head/charset). The next lessons cover title, description, and canonical in an SEO light.

## The Body: Real Text, Real Structure

Search engines are built to read **text.** Help them:

- **Put your content in real text,** not in images. If you must use an image of text, add alt text that repeats the words.
- **Use one `h1`** that says what the page is about, and `h2` to `h6` for sections. See [Headings and SEO](./headings-and-seo).
- **Use semantic elements** such as `main`, `nav`, `article`, `section`, and `footer`. See [Semantic HTML and SEO](./semantic-html-and-seo).
- **Use lists and tables** for lists and data. They carry meaning.
- **Write one clear topic per page,** answering a real question. If you cover two different topics, make two pages.
- **Write for people.** Repeating a keyword over and over helps no one.

## Links Search Engines Can Follow

Crawlers discover your site by **following links.** A link is only a link if it is a real `a` element with an `href`.

```html
<!-- Good: a crawler can follow it, and so can a keyboard -->
<a href="/recipes/rye-bread">Rye bread recipe</a>

<!-- Poor: no href, so no link -->
<a onclick="go('/recipes/rye-bread')">Rye bread recipe</a>
<span onclick="go('/recipes/rye-bread')">Rye bread recipe</span>
```

- **Write descriptive link text.** "Rye bread recipe" says far more than "click here." It helps people, screen readers, and crawlers. See also [Accessible Links](/tutorials/html/accessibility/accessible-links).
- **Link between your own pages,** so no page is an "orphan" with nothing pointing at it. See [Internal Links](/tutorials/html/links-and-navigation/internal-links).
- **Choose `rel` values for outside links** when it matters: `rel="nofollow"` for links you do not want to vouch for, `rel="sponsored"` for paid links, and `rel="ugc"` for links in user comments. See [External Links](/tutorials/html/links-and-navigation/external-links).

## Images

- **Add useful alt text** that describes what the image shows. It helps accessibility and image search. See [Alt Text](/tutorials/html/accessibility/alt-text).
- **Give images clear file names,** such as `rye-bread-loaf.jpg` rather than `IMG_0042.jpg`.
- **Set `width` and `height`,** so the page does not jump while images load.
- **Use the right size and format.** Huge images slow the page. See [Responsive Images](/tutorials/html/images/responsive-images).
- **Lazy-load images below the fold** with `loading="lazy"`, but not your main image at the top. See [Lazy Loading](/tutorials/html/images/lazy-loading).

Some friendly habits:

- **Use words that describe the page:** `/recipes/rye-bread`.
- **Use lowercase and hyphens,** not spaces or underscores.
- **Keep them short and stable.** If an address must change, **redirect** the old one to the new one with a permanent redirect (301).
- **Avoid long strings of parameters** and anything private, such as session ids or emails.
- **Use https** everywhere.

## Mobile and Speed Basics

Most searches today happen on phones, and search engines largely look at the **mobile version** of your page. A page that is slow or hard to use on a phone is hard to love. Several basics live in your HTML:

- **The viewport meta tag,** so the page scales to the device.
- **Responsive layouts,** which are covered in your CSS lessons.
- **`width` and `height` on images,** to avoid layout jumps.
- **`defer` or `async` on scripts,** so they do not block the page. See [Script](/tutorials/html/html-head/script).
- **Small, compressed images** and few heavy third-party scripts.
- **Large tap targets** and readable text.

Speed and page experience are only **part** of ranking, but slow pages also lose visitors, whatever the ranking.

## JavaScript and SEO

Many modern sites build pages in the browser with JavaScript. That can work, but it adds risk:

- The crawler may see an **empty page** at first, and fill in the content later.
- Some search engines, social apps, and other bots **do not run scripts.**
- Content that only loads after a click or scroll may **never be seen.**
- Links created without `href` are **not followed.**

How to stay safe:

1. **Send the important content in the first HTML** (server-side rendering or static generation). Docusaurus, which powers this site, builds static HTML pages for exactly this reason.
2. **Use real links** for navigation, even in a single-page app.
3. **Give each page its own address,** title, and description.
4. **Test with "view source"** and with Search Console's URL inspection tool.

## Status Codes, Sitemaps, and robots.txt

A few things sit outside your HTML, but they affect everything:

- **Status codes.** A working page returns 200. A moved page returns a redirect (301 for permanent). A missing page should return **404.** A page that says "not found" but returns 200 is called a **soft 404,** and confuses search engines.
- **A sitemap** (usually `sitemap.xml`) lists the pages you want found. It helps, but it does not guarantee they are indexed.
- **robots.txt** is a file at the top of your site that asks crawlers which paths to avoid. It controls **crawling,** not whether a page can appear in results. The `robots` meta tag is covered in [Robots Meta](./robots-meta).

## Good Habits

- **Put the important content in the HTML.**
- **Give each page one purpose,** a unique title, and a description.
- **Use real links with descriptive text.**
- **Describe images,** and give them sizes.
- **Keep addresses clean and stable,** with redirects when they change.
- **Test on mobile,** and on slow connections.
- **Check with "view source" and Search Console.**
- **Be patient.** SEO takes time.

## SEO Basics Checklist

- [ ] `doctype`, `lang`, `charset`, and `viewport` are set
- [ ] The title is unique and descriptive
- [ ] There is a meta description
- [ ] There is one `h1`, and headings follow an outline
- [ ] Main content is real text inside `main`
- [ ] Links are real `a` elements with `href`, and descriptive text
- [ ] Images have alt text, clear names, and width and height
- [ ] Addresses are clean, lowercase, and stable, and use https
- [ ] Old addresses redirect with 301
- [ ] Missing pages return 404

- [ ] A sitemap exists, and robots.txt does not block what you want found

## Security and Privacy Notes

- **Do not put private information in URLs.** Emails, tokens, and session ids in an address end up in browser history, logs, and shared links, and can be indexed.
- **Check what is public.** Staging sites, test pages, backups, and admin screens can be found by crawlers if they are reachable. Protect them with a login, and not only with `noindex` or robots.txt.
- **`robots.txt` is public.** Anyone can read it, so do not list secret paths in it.
- **Use https.** It protects visitors, and search engines prefer it.
- **Watch out for hidden text and cloaking,** showing crawlers something different from what people see. It is against search guidelines, and it can look like a security problem.
- **Use `rel="ugc"` or `rel="nofollow"`** on links added by visitors, and moderate comments, to limit spam.
- **Respect privacy laws** if you add analytics or tracking. This is not legal advice, so check the rules that apply to you.

## Global Audience Notes

- **Set `lang` correctly,** and write each page in one main language. See [International SEO](./international-seo).
- **Use UTF-8** so names and scripts from any language display correctly.
- **Addresses can contain non-English words.** Many sites keep addresses in a simple form (transliterated or English) for easier sharing. Whatever you choose, be consistent.
- **Mobile and slow networks matter most** in many regions. Test on modest phones.
- **Search engines differ by country.** Check the guidelines of the ones your readers use.
- **Date, currency, and units** should match your readers' region.

## Common Mistakes to Avoid

1. **Hiding content in scripts,** so "view source" shows an empty page.
2. **Links with no `href`,** or buttons used as links.
3. **Text inside images** with no alt text.
4. **Generic titles** such as "Home" on every page.
5. **No `h1`,** or several of them, or headings used only for style.
6. **Link text like "click here."**
7. **Messy, long, changing addresses,** with no redirects.
8. **Missing viewport tag,** so phones show a tiny page.
9. **Large, unsized images.**
10. **Using `noindex` by accident,** often left over from testing.
11. **Soft 404 pages** that say "not found" but return 200.
12. **Never checking** what search engines actually see.

## Quick Recap

Click a question to check your answer.

<details>
<summary>1. What does a crawler first receive from your page?</summary>

The raw HTML from the server. It reads the head, the text and structure, the links, the images, and any structured data. Running JavaScript is a later, less certain step.

</details>

<details>
<summary>2. What are the head essentials?</summary>

The doctype, the `lang` attribute, the charset, the viewport meta tag, a title, a meta description, and, when needed, a canonical link.

</details>

<details>
<summary>3. What makes a link something a crawler can follow?</summary>

It must be an `&lt;a&gt;` element with an `href`. Click handlers on other elements are not links.

</details>

<details>
<summary>4. Why set width and height on images?</summary>

So the browser can reserve space for them, and the page does not jump around while images load.

</details>

<details>
<summary>5. How can JavaScript hurt SEO?</summary>

Content that only appears after scripts run can be missed or delayed, and some crawlers do not run scripts. Send important content in the HTML, and use real links.

</details>

<details>
<summary>6. What is a soft 404?</summary>

A page that tells people "not found" but returns a 200 (success) status code. Return a real 404 for missing pages.

</details>

## Key Takeaways

- **Put the important content in the HTML.** "View source" shows what a crawler gets first.
- **Every page needs the head essentials,** a unique title, and a description.
- **Use real text, headings, and semantic elements** so the structure is clear.
- **Real links with descriptive text** help crawlers, people, and screen readers.
- **Describe images,** give them sizes, and name them clearly.
- **Keep addresses clean and stable,** and redirect when they change.
- **Mobile, speed, and JavaScript choices** can help or hurt.
- **Check your pages,** and be patient.
