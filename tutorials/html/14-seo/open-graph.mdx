export function escAttr(value) {
  return String(value).replace(/&/g, "&amp;").replace(/"/g, "&quot;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

export const ogTypes = ["website", "article", "book", "profile", "music.song", "music.album", "music.playlist", "music.radio_station", "video.movie", "video.episode", "video.tv_show", "video.other"];

export function OgCheckerInner() {
  const [which, setWhich] = useState("messy");
  const [html, setHtml] = useState(ogSamples.messy);
  const doc = new DOMParser().parseFromString("<!doctype html>" + html, "text/html");
  const issues = [];
  const notes = [];
  const wrongName = Array.from(doc.querySelectorAll('meta[name^="og:"]'));
  wrongName.forEach((el) => issues.push("<meta name=\"" + el.getAttribute("name") + "\"> uses name. Open Graph tags should use property=\"" + el.getAttribute("name") + "\"."));
  const tags = Array.from(doc.querySelectorAll('meta[property]'));
  const byProp = {};
  tags.forEach((el) => {
    const key = el.getAttribute("property").trim();
    (byProp[key] = byProp[key] || []).push((el.getAttribute("content") || "").trim());
  });
  ["og:title", "og:type", "og:image", "og:url"].forEach((key) => {
    if (!byProp[key] && !(key === "og:title" && wrongName.some((el) => el.getAttribute("name") === key))) issues.push("Missing " + key + ", one of the four essential properties.");
  });
  ["og:title", "og:type", "og:url", "og:description"].forEach((key) => {
    if (byProp[key] && byProp[key].length > 1) issues.push(key + " appears " + byProp[key].length + " times. Use it once.");
  });
  if (byProp["og:description"] && byProp["og:description"][0] === "") issues.push("og:description is empty.");
  if (!byProp["og:description"]) notes.push("No og:description. Some apps fall back to the page's meta description, and some show nothing.");
  if (byProp["og:type"] && byProp["og:type"][0] && ogTypes.indexOf(byProp["og:type"][0]) === -1) issues.push("og:type \"" + byProp["og:type"][0] + "\" is not a standard type. Common ones are website and article.");
  (byProp["og:image"] || []).forEach((value) => {
    if (!/^https?:\/\//i.test(value)) issues.push("og:image \"" + value + "\" is not a full address. Use https://.");
    else if (/\.svg(\?|$)/i.test(value)) issues.push("og:image \"" + value + "\" is an SVG. Most share previews do not accept SVG. Use JPG, PNG, or WebP.");
  });
  if ((byProp["og:image"] || []).length > 1) notes.push("There are " + byProp["og:image"].length + " og:image tags. The first is usually the one used, so put the best one first.");
  if (byProp["og:image"] && !byProp["og:image:alt"]) notes.push("No og:image:alt. Add a text description of the image.");
  if (byProp["og:image"] && (!byProp["og:image:width"] || !byProp["og:image:height"])) notes.push("No og:image:width and og:image:height. They help apps show the card on the first share, before they have downloaded the image.");
  const url = byProp["og:url"] && byProp["og:url"][0];
  if (url) {
    if (!/^https:\/\//i.test(url)) issues.push("og:url \"" + url + "\" should be a full https address.");
    if (/[?&](utm_|fbclid|gclid|sessionid)/i.test(url)) issues.push("og:url contains tracking or session parameters. Use the clean address.");
    const canonical = doc.querySelector('link[rel~="canonical"]');
    if (canonical && canonical.getAttribute("href") && canonical.getAttribute("href").trim() !== url) notes.push("og:url differs from the canonical link. They should normally match. See Canonical Links.");
  }
  const articleTags = Object.keys(byProp).filter((key) => key.indexOf("article:") === 0);
  const type = byProp["og:type"] && byProp["og:type"][0];
  if (articleTags.length > 0 && type !== "article") notes.push("There are article: tags, but og:type is not \"article\".");
  (byProp["article:published_time"] || []).concat(byProp["article:modified_time"] || []).forEach((value) => {
    if (!/^\d{4}-\d{2}-\d{2}(T\S+)?$/.test(value)) issues.push("The date \"" + value + "\" is not in ISO format (YYYY-MM-DD).");
  });
  if (!byProp["og:site_name"]) notes.push("No og:site_name. Some apps show it above the title.");
  if (!byProp["og:locale"]) notes.push("No og:locale. Add one, such as en_US, if your page is not in English or you serve several languages.");
  const choose = (value) => { setWhich(value); setHtml(ogSamples[value]); };
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>Which page do you want to check?</legend>
        <label style={{marginRight: "1rem"}}><input type="radio" name="ogc-sample" checked={which === "messy"} onChange={() => choose("messy")} /> A messy page</label>
        <label><input type="radio" name="ogc-sample" checked={which === "fixed"} onChange={() => choose("fixed")} /> The fixed version</label>
      </fieldset>
      <p style={{margin: "0 0 0.5rem"}}>
        <label htmlFor="ogc-html">HTML head (edit it freely)</label><br />
        <textarea id="ogc-html" rows={12} value={html} onChange={(event) => setHtml(event.target.value)} style={{width: "100%", fontFamily: "monospace"}} />
      </p>
      <p role="status" style={{margin: "0 0 0.4rem"}}><strong>{issues.length}</strong> problems, <strong>{notes.length}</strong> suggestions.</p>
      {issues.length === 0 && notes.length === 0 ? <p style={{margin: 0}}>Nothing to report.</p> : null}
      <ul style={{margin: 0, paddingLeft: "1.2rem"}}>
        {issues.map((issue, index) => (<li key={"i" + index}><strong>Problem:</strong> {issue}</li>))}
        {notes.map((note, index) => (<li key={"n" + index}><strong>Suggestion:</strong> {note}</li>))}
      </ul>
      <p style={{margin: "0.6rem 0 0", fontSize: "0.9em", color: "#64748b"}}>
        This reads the HTML only. It cannot fetch your image, check its size, or see what an app has cached. Test the live page with the sharing debuggers.
      </p>
    </div>
  );
}

If you do not add them, apps take a guess, and the guess is often a blurry image (or none at all) and the wrong words. If you add them, **you** decide how your page introduces itself to the world.

## What Open Graph Is

**Open Graph** is a small set of tags first created by Facebook, and now read by many other apps: social networks, chat apps, collaboration tools, and link-preview services. When someone shares your link, the app **fetches your page,** reads the tags, and draws a card.

- **It is not for search rankings.** Open Graph does not directly help your search position. Its job is to make shared links look good, and a good-looking link gets more clicks.
- **It uses `property`, not `name`.** Open Graph tags are written `property="og:title"`. (Many parsers also accept `name`, but follow the standard.)
- **It lives in the head.** Like all meta tags.

## The Four Essential Tags

| Tag | What it says | Example |
| :-- | :-- | :-- |
| `og:title` | The headline of the card | Easy No-Knead Rye Bread |
| `og:type` | What kind of thing this is | `article` or `website` |
| `og:image` | The picture for the card | A full https address |
| `og:url` | The main address of the page | The canonical address |

Add these, and you have the foundation. Then add the helpful extras:

| Extra | Why |
| :-- | :-- |
| `og:description` | One or two lines under the title |
| `og:image:alt` | A text description of the picture, for accessibility |
| `og:image:width` and `og:image:height` | Helps apps lay out the card on the first share |
| `og:site_name` | The name of your site, shown above or below the title |
| `og:locale` | The language and region, such as `en_US` or `fr_FR` |
| `article:published_time` | The date, for articles (ISO format) |
| `article:author` and `article:section` | More about an article |

### Choosing og:type

- **`website`** for your home page and general pages.
- **`article`** for blog posts, recipes, and news. It unlocks the `article:` tags.
- **Others,** such as `book`, `profile`, `video.movie`, and `music.song`, exist for those kinds of content.

If you are unsure, `website` is a safe default. Some apps ignore the type altogether.

## Choosing a Share Image

The image is the part of the card that people notice first, and the one that most often goes wrong.

- **Use a wide image,** about **1200 by 630 pixels** (roughly 1.91 to 1). This is a common size that many apps display as a large card. Check each platform's current guidance, as sizes change.
- **Use JPG, PNG, or WebP.** Most apps do not accept SVG.
- **Keep the file size sensible.** Large files may be skipped by some apps, and are slow on mobile data. A few hundred kilobytes is usually plenty.
- **Make it a full address,** starting with `https://`. Relative paths such as `/img/rye.jpg` do not work, because the app is not on your site.
- **The image must be public.** If it is behind a login, the app cannot fetch it.
- **Keep the important parts in the middle.** Cards crop images in different ways.
- **Put text in the image carefully.** It must be large and short, because the card is small. It should also be in the page's text somewhere, for accessibility.
- **Always add `og:image:alt`.**
- **Do not use the same image for every page.** A unique image per page makes shared links much more recognisable.

## Full Addresses and Clean URLs

- **All addresses must be absolute,** with `https://`. This goes for `og:image` and `og:url`.
- **`og:url` should be the canonical address,** with no tracking parameters, so that every share of the same page is counted and displayed as one. See [Canonical Links](./canonical-links).
- **Do not point `og:url` at a redirect,** or at a different page.

## What Apps Do When Tags Are Missing

Most apps have a **fallback** chain:

- No `og:title`: use the page's `<title>`.
- No `og:description`: use the meta description, or text from the page.
- No `og:image`: look for another image on the page, or show a text-only card.

That is why a page with no Open Graph tags still gets a preview, and why the preview is so often poor. Fallbacks vary by app, and are not guaranteed.

## Open Graph, Title, and Meta Description

These often hold similar text, but they are **different tags for different audiences:**

| Tag | Audience | Notes |
| :-- | :-- | :-- |
| `<title>` | People choosing a search result | Include your brand. See [Title (SEO)](./title-seo) |
| `meta name="description"` | People reading a search snippet | See [Meta Description](./meta-description) |
| `og:title` | People seeing a shared link | Can be shorter and friendlier, and often leaves out the brand |
| `og:description` | People seeing a shared link | Can be warmer, or add a hook |

They should agree on the topic, but they do not have to match word for word.

## Scrapers Do Not Run Your JavaScript

When an app fetches your link, it usually downloads the HTML and reads it **without running scripts.** So:

- **Open Graph tags must be in the HTML the server sends,** not added later by JavaScript.
- **Single-page apps** often fail here. Everyone sees the same default card, whatever page is shared. The fix is server-side rendering, static generation, or a pre-rendering service.
- **Docusaurus** builds static HTML for each page, so tags in the head are in the file when it is served. Always check the built page with "view source."

## Caching: Why the Preview Will Not Update

Apps **save** your preview after the first share. If you change your tags or image afterwards, the old card may keep appearing, sometimes for days.

- **Use a debugger tool to refresh it.** For example, Facebook's Sharing Debugger and LinkedIn's Post Inspector let you fetch a page again and clear the old card. (Tool names change, so check the platform's help.)
- **Use a new image address** when you change a picture (for example, add a version number to the file name). Apps cache by address.
- **Chat apps** cache too, and some offer no way to refresh. Test your page **before** you share it widely.

## Security and Privacy Notes

- **Escape everything.** If a title or description comes from a database or a user, a stray quote or angle bracket can break your tags, and may inject content. Escape values before you put them in `content="..."`. The builder above does this.
- **Do not leak private pages.** If someone shares a link to a private or logged-in page, the app may fetch it, and show the title and image to everyone who sees the card. Do not put private details in the tags of private pages. Use a generic title and image for pages behind a login.
- **Pictures can reveal more than you expect.** A share image can include names, faces, and locations. Get permission, and check what is in the picture. Also strip hidden location data (EXIF) from photos.
- **Do not mislead.** A card that says one thing while the page says another is deceptive, and could be treated as spam or a scam by the platform.
- **Do not copy another brand's name or look** in your tags. This is not legal advice.
- **Apps may fetch your page and image** from their own servers. This shows in your logs. It is normal, but if you restrict access by address, allow the platforms you use.
- **Watch for tampering.** A hacked page can have its tags changed, so that shares show spam. Check your live tags from time to time.

## Global Audience Notes

- **Set `og:locale`** to the page's language and region (for example `fr_FR`). For other languages of the same page, add `og:locale:alternate` tags.
- **Write the tags in the language of the page.** Each translation needs its own title, description, and ideally its own image.
- **Text in images is not translated.** If your share image has words in it, make a version per language, or keep the image free of text.
- **Right-to-left languages** need an image and text that read correctly in the card. Test them.
- **Titles and descriptions get cut at different lengths** in different scripts, so check the preview in each language.
- **Platforms differ by country.** The app that matters in one country may not exist in another. Open Graph is widely read, but check where your readers share.
- **Slow networks:** a large image may not load in time on a poor connection, and the card may appear without it. Keep files small.

## Common Mistakes to Avoid

1. **No Open Graph tags at all.**
2. **A relative image address.**
3. **A tiny or square image** that gets cropped or shrunk.
4. **An SVG** as the share image.
5. **The same image on every page.**
6. **Using `name` instead of `property`.**
7. **Tags added only by JavaScript.**
8. **`og:url` with tracking parameters,** or different from the canonical.
9. **A private or login-protected image.**
10. **A stale preview** that is never refreshed.
11. **Missing `og:image:alt`.**
12. **Never testing** the card before sharing.
