import {useState} from 'react';

export function LayoutBuilderDemo() {
  const [parts, setParts] = useState({header: true, nav: true, aside: true, footer: true});
  const toggle = (key) => setParts((old) => ({...old, [key]: !old[key]}));
  const roles = {header: "banner", nav: "navigation", aside: "complementary", footer: "contentinfo"};
  const box = (bg, extra) => ({background: bg, border: "1px solid #94a3b8", borderRadius: "6px", padding: "0.5rem 0.75rem", fontSize: "0.9em", ...extra});
  const landmarks = [];
  if (parts.header) landmarks.push("banner (header)");
  if (parts.nav) landmarks.push("navigation (nav)");
  landmarks.push("main (main)");
  if (parts.aside) landmarks.push("complementary (aside)");
  if (parts.footer) landmarks.push("contentinfo (footer)");
  const warnings = [];
  if (!parts.header) warnings.push("No banner: visitors cannot jump to the site header or logo.");
  if (!parts.nav) warnings.push("No navigation: screen reader users cannot list your menus.");
  if (!parts.footer) warnings.push("No contentinfo: the copyright and legal links have no landmark.");
  return (
    <div>
      <fieldset style={{margin: "0 0 0.6rem"}}>
        <legend>Include these parts</legend>
        {Object.keys(parts).map((key) => (
          <label key={key} style={{display: "inline-block", marginRight: "1rem"}}>
            <input type="checkbox" checked={parts[key]} onChange={() => toggle(key)} /> {key}
          </label>
        ))}
        <label style={{display: "inline-block", opacity: 0.7}}>
          <input type="checkbox" checked disabled /> main (always)
        </label>
      </fieldset>
      <div style={{display: "grid", gap: "0.4rem", gridTemplateColumns: parts.aside ? "1fr 9rem" : "1fr"}}>
        {parts.header && <div style={box("#dbeafe", {gridColumn: "1 / -1"})}><strong>header</strong> · role: {roles.header}</div>}
        {parts.nav && <div style={box("#dcfce7", {gridColumn: "1 / -1"})}><strong>nav</strong> · role: {roles.nav}</div>}
        <div style={box("#fef9c3", {minHeight: "5rem"})}><strong>main</strong> · role: main<br />article, sections…</div>
        {parts.aside && <div style={box("#fae8ff")}><strong>aside</strong> · role: {roles.aside}</div>}
        {parts.footer && <div style={box("#e2e8f0", {gridColumn: "1 / -1"})}><strong>footer</strong> · role: {roles.footer}</div>}
      </div>
      <p style={{margin: "0.6rem 0 0.2rem"}}><strong>A screen reader's landmark list:</strong> {landmarks.join(", ")}</p>
      <ul role="status" style={{margin: 0, paddingLeft: "1.25rem", color: "#b91c1c"}}>
        {warnings.map((text) => (<li key={text}>{text}</li>))}
      </ul>
    </div>
  );
}

export function ChecklistDemo() {
  const items = [
    "The page has a lang attribute and a unique title",
    "There is exactly one visible main element",
    "A header and footer at page level",
    "Every nav has a clear name (aria-label) if there are several",
    "There is one h1, and headings do not skip levels",
    "Each section has a heading",
    "A skip link is the first focusable element",
    "The DOM order matches the visual order",
    "Landmarks are not wrapped in extra landmark roles",
    "The page still makes sense with CSS turned off",
  ];
  const [checked, setChecked] = useState({});
  const done = items.filter((_, index) => checked[index]).length;
  const percent = Math.round((done / items.length) * 100);
  let verdict = "Keep going: tick each item you have checked.";
  if (done === items.length) verdict = "Excellent. Now test with a keyboard and a screen reader.";
  else if (done >= 7) verdict = "Nearly there.";
  return (
    <form onSubmit={(event) => event.preventDefault()}>
      <fieldset>
        <legend>Page layout checklist</legend>
        {items.map((text, index) => (
          <p key={text} style={{margin: "0.2rem 0"}}>
            <label>
              <input type="checkbox" checked={!!checked[index]} onChange={() => setChecked((old) => ({...old, [index]: !old[index]}))} /> {text}
            </label>
          </p>
        ))}
      </fieldset>
      <p style={{margin: "0.6rem 0 0.2rem"}}>
        <progress max="100" value={percent} aria-label="Checklist progress" style={{width: "12rem", verticalAlign: "middle"}} /> {done} of {items.length}
      </p>
      <p role="status" style={{margin: 0}}>{verdict}</p>
    </form>
  );
}

Over the last lessons you met each semantic element on its own: `header`, `nav`, `main`, `section`, `article`, `aside`, `footer`, `address`, `details`, and `dialog`. Now it is time to **put them together** into a real page.

A good page layout is like a well-planned building: a clear entrance, signs on every floor, one main hall, and a place for the small rooms on the side. Visitors, search engines, and assistive technology can all find their way around, because the structure is **in the markup**, not only in the paint.

Read the skeleton from top to bottom:

| Part | Role | Notes |
| :-- | :-- | :-- |
| Skip link | None | The first focusable item, jumping to `main` |
| Page `header` | **banner** | Logo, site name, and often the main `nav` |
| `nav` | **navigation** | Named with `aria-label`, because there are two on the page |
| `main` | **main** | One per page, with the page's unique content |
| `article` | Generic (article) | A self-contained piece, with its own header |
| `section` | **region** only when named | Each has a heading |
| `aside` | **complementary** | Related, not essential |
| Page `footer` | **contentinfo** | Copyright, legal and contact links |

Two `header` elements appear above. The first is the **page header** (a banner). The second sits inside an `article`, so it is only the article's own heading group. See [Header](./header) for the full rules.

## The Skip Link

Keyboard users have to tab through your whole menu on **every page** before they reach the content. A **skip link** is the shortcut. It is the **first focusable element**, and it jumps to `main`.

```html
<a class="skip-link" href="#main">Skip to main content</a>
…
<main id="main" tabindex="-1">
```

```css
.skip-link {
  position: absolute;
  inset-inline-start: 0.5rem;
  top: -4rem;              /* off screen until focused */
  background: #1d4ed8;
  color: #fff;
  padding: 0.5rem 1rem;
  z-index: 10;
}

.skip-link:focus {
  top: 0.5rem;             /* slides into view */
}
```

Do **not** use `display: none` on a skip link, because then nobody can focus it. The `tabindex="-1"` on `main` lets the browser move the focus into the content after the jump. See [Main](./main) for details.

## Heading Outline

Headings form the **table of contents** that screen reader users navigate by. Plan them as a simple outline:

```text
h1  Sourdough for Beginners          ← one per page
  h2  Make a starter
  h2  Bake the loaf
    h3  Scoring the dough
  h2  Tools you need                  ← the aside
```

- **One `h1`** that names the page (usually matching the `title`).
- **Do not skip levels** (an `h2` followed by an `h4`).
- **Choose levels by importance,** not by the size you want. Use CSS to change the size.
- **Every `section` has a heading**, and the `aside` has one too.
- The "document outline algorithm" is **not implemented** by browsers, so nesting `section` elements does not change heading levels. See [Section](./section).

## Laying It Out with CSS Grid

The elements are just boxes. CSS decides where they sit. A common layout is a header on top, content and a sidebar in the middle, and a footer at the bottom:

<Tabs>
<TabItem value="code" label="HTML Code" default>

```css title="style.css"
body {
  min-height: 100vh;
  margin: 0;
  display: grid;
  grid-template-rows: auto 1fr auto;   /* header, content, footer */
}

main {
  display: grid;
  grid-template-columns: minmax(0, 1fr) 16rem;   /* article, aside */
  gap: 2rem;
  padding: 1rem;
  max-width: 70rem;
  margin-inline: auto;
}

/* One column on small screens */
@media (max-width: 48rem) {
  main {
    grid-template-columns: 1fr;
  }
}
```

### Keep the Source Order Meaningful

CSS can move boxes around, but **screen readers and the Tab key follow the HTML order**, not the visual order. Watch out for:

- `order`, `grid-area` placement, or `flex-direction: row-reverse` that makes the **visual order differ** from the DOM order. Keyboard focus then seems to jump around.
- A sidebar placed **before** `main` in the HTML, so every visitor reads it first. Put `aside` **after** the article it relates to, and let the layout position it.
- Content that only makes sense in a certain visual position ("see the box on the right"). Describe it by name instead: "see the tools list".

The rule: **write the HTML in the order you want it read, and use CSS only to arrange it.**

## Common Layout Patterns

| Layout | Structure |
| :-- | :-- |
| **Blog post** | Header, nav, main with one `article`, an optional `aside`, footer |
| **Landing page** | Header, nav, main with several named `section` elements (hero, features, pricing), footer |
| **Dashboard** | Header, nav (sidebar menu), main with sections and a `dialog` for settings, no footer needed |
| **Docs site** | Header, nav (docs menu), main with `article`, `aside` with the page's table of contents, footer |
| **Web app** | A stable shell (header, nav, main, footer), with only `main`'s content changing, and focus moved on route change |

In all of them, the **same shell** repeats on every page, so people learn it once.

## Page Quality Beyond Landmarks

A semantic page is more than tags:

- `lang` on the `html` element, and `lang` on foreign phrases.
- A **unique, descriptive `title`** per page. It is the first thing a screen reader announces.
- A **viewport meta tag**, so mobile browsers do not zoom out.
- A **visible focus style** on every link and button.
- **Sufficient colour contrast**, and a layout that works at 200% zoom.
- **Real links and buttons**, as explained in [Why Semantic HTML](./why-semantic-html).
- **Landmark labels** only where you need to tell similar ones apart.

## Accessibility at a Glance

- **Use native landmarks** (`header`, `nav`, `main`, `aside`, `footer`). Do not repeat their roles in ARIA.
- **Exactly one visible `main`.**
- **Name repeated landmarks** (two `nav` elements, two `aside` elements).
- **Add a skip link,** and make it visible on focus.
- **Keep the DOM order logical,** and match the visual order.
- **Do not hide** the page's structure from assistive technology with `aria-hidden` on a landmark.
- **Test by keyboard,** by landmark list in a screen reader, and with CSS turned off.

## Security and Privacy Notes

- **Layout is not protection.** Hiding an admin `section` or a `footer` link with CSS does not hide it from the page source. Enforce permissions on the server.
- **Do not put personal data in landmark labels or headings** (for example "Account 4821 overview") where it can appear in tab titles, history, and screen shares.
- **Escape any user-supplied text** you place in headings, `title`, or `aria-label`.
- **Third-party widgets** (ads, comments, embeds) inside `aside` or `iframe` should be clearly titled and restricted with `sandbox` and `rel` values such as `noopener`, so they cannot change your page or capture its focus.

## Global Audience Notes

- **Set `lang` and `dir`** on `html`, and use **logical CSS properties** (`margin-inline`, `padding-inline-start`, `inset-inline-start`), so the whole layout mirrors in right-to-left languages.
- **Allow text to grow,** since translations can be 30% longer or more. Avoid fixed-height headers and buttons.
- **Keep the page light.** Many visitors use slow or metered connections, and old phones. Semantic HTML is small and fast by nature.
- **Use plain words in landmark labels** ("Main", "Footer"), and translate them with the page.
- **Do not rely on hover or wide screens.** Test at narrow widths and on touch.

## Common Mistakes to Avoid

1. **A page built only from `div` elements** (div soup).
2. **More than one visible `main`,** or none.
3. **Page-level `header` and `footer` placed inside `main`.**
4. **Two unnamed `nav` elements.**
5. **Skipping the skip link,** or hiding it with `display: none`.
6. **Several `h1` elements, or skipped heading levels.**
7. **A `section` with no heading,** used as a styling wrapper.
8. **Putting the sidebar before the content** in the HTML.
9. **Reordering with CSS** so keyboard focus jumps around.
10. **Adding `role="main"` to `main`,** or other redundant ARIA.
11. **Forgetting `lang` and a useful `title`.**
12. **Never testing** with the keyboard, a screen reader, or CSS turned off.

## Quick Recap

Click a question to check your answer.

<details>
<summary>1. What are the main parts of a semantic page layout?</summary>

A page `header` (banner), `nav` (navigation), one `main`, optional `aside` (complementary), and a page `footer` (contentinfo), plus `article` and `section` elements inside `main` to organise the content.

</details>

<details>
<summary>2. Why add a skip link?</summary>

It lets keyboard and screen reader users jump past the repeated menu straight to the main content, saving many key presses on every page. It must be the first focusable element and visible when focused.

</details>

<details>
<summary>3. Does CSS change the order screen readers and the Tab key follow?</summary>

No. They follow the HTML source order. Write the HTML in the reading order you want, and use CSS only to arrange the boxes. Avoid reordering that makes the visual and DOM orders differ.

</details>

<details>
<summary>4. When must a `nav` or `aside` be given a name?</summary>

When there is more than one of the same landmark on the page, so users can tell them apart in a landmark list. Use `aria-label` or `aria-labelledby`.

</details>

<details>
<summary>5. How should headings be arranged on a page?</summary>

One `h1` for the page, then `h2` for main sections and `h3` below them, with no skipped levels. Choose levels by meaning and use CSS for size.

</details>

<details>
<summary>6. How do you test a layout?</summary>

Use the keyboard alone, list the landmarks and headings with a screen reader or browser tool, turn CSS off to check the source order, and zoom to 200% and a narrow width.

</details>

## Key Takeaways

- A semantic page is a **skeleton of landmarks**: header, nav, main, aside, footer, with `article` and `section` inside `main`.
- Use **one visible `main`**, a **skip link**, and **named** repeated landmarks.
- Plan the **heading outline** first: one `h1`, no skipped levels, and a heading for every `section`.
- Use **CSS grid** for the layout, but keep the **HTML order** meaningful, because assistive technology and the keyboard follow it.
- Do not add ARIA roles that native elements already provide.
- Layout hides nothing from the page source, so enforce permissions on the server.
- Use logical CSS properties, set `lang` and `dir`, and test on narrow, slow, and right-to-left setups.
- Always test with the keyboard, a screen reader, and CSS turned off.
