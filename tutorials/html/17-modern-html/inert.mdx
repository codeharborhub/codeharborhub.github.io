import {useState, useEffect, useRef} from 'react';

export function probeFocus(root) {
  const items = Array.from(root.querySelectorAll("[data-probe]"));
  const results = items.map(function (el) {
    el.focus();
    return {label: el.getAttribute("data-probe"), reached: document.activeElement === el};
  });
  if (document.activeElement && document.activeElement !== document.body) document.activeElement.blur();
  return results;
}

export function ProbeList(props) {
  if (props.results === null) return null;
  return (
    <ul aria-live="polite" style={{margin: "0.5rem 0"}}>
      {props.results.map(function (r) {
        return (
          <li key={r.label} style={{color: r.reached ? "#c0392b" : "#1e8449"}}>
            <strong>{r.label}:</strong> {r.reached ? "accepted keyboard focus" : "refused keyboard focus"}
          </li>
        );
      })}
    </ul>
  );
}

export function FocusLab() {
  const areaRef = useRef(null);
  const [mode, setMode] = useState("normal");
  const [clicks, setClicks] = useState(0);
  const [probe, setProbe] = useState(null);
  const [supported, setSupported] = useState(null);
  useEffect(function () {
    setSupported(typeof HTMLElement !== "undefined" && "inert" in HTMLElement.prototype);
  }, []);
  useEffect(function () {
    const el = areaRef.current;
    if (!el) return;
    if ("inert" in el) el.inert = mode === "inert";
    if (mode === "aria") el.setAttribute("aria-hidden", "true");
    else el.removeAttribute("aria-hidden");
    setProbe(null);
  }, [mode]);
  const notes = {
    normal: "Nothing is switched off. Every control accepts focus.",
    inert: "The whole area refuses focus and clicks, and is hidden from screen readers.",
    aria: "Screen readers are told to skip the area, but the keyboard can still land inside it. That mismatch is the trap."
  };
  return (
    <div style={{padding: "1rem"}}>
      {supported === false && (
        <p style={{color: "#c0392b", fontWeight: 600}}>This browser does not support inert, so the inert option will have no effect here. That is what a visitor with an older browser would get.</p>
      )}
      <fieldset style={{border: "none", padding: 0, margin: "0 0 0.75rem"}}>
        <legend style={{fontWeight: 600}}>Switch the area</legend>
        {[["normal", "Normal"], ["inert", "inert"], ["aria", "aria-hidden=\"true\" only"]].map(function (o) {
          return <label key={o[0]} style={{marginRight: "1rem"}}><input type="radio" name="fl-mode" checked={mode === o[0]} onChange={function () { setMode(o[0]); }} /> {o[1]}</label>;
        })}
      </fieldset>
      <div ref={areaRef} style={{border: "1px solid rgba(128,128,128,0.6)", borderRadius: 6, padding: "0.75rem", opacity: mode === "inert" ? 0.5 : 1}}>
        <p style={{margin: "0 0 0.5rem", fontWeight: 600}}>Shop area</p>
        <p style={{margin: "0 0 0.5rem"}}>
          <button type="button" data-probe="Buy button" onClick={function () { setClicks(clicks + 1); }}>Buy now</button>{" "}
          <a href="#help-demo" data-probe="Help link" onClick={function (e) { e.preventDefault(); }}>Help</a>
        </p>
        <p style={{margin: "0 0 0.5rem"}}>
          <label htmlFor="fl-name">Name</label>{" "}
          <input id="fl-name" type="text" data-probe="Name field" />
        </p>
        <label><input type="checkbox" data-probe="Newsletter checkbox" /> Send me the newsletter</label>
      </div>
      <p style={{margin: "0.75rem 0 0.25rem"}}>Clicks received by the Buy button: <strong>{clicks}</strong> (try clicking it in each mode)</p>
      <button type="button" onClick={function () { setProbe(probeFocus(areaRef.current)); }}>Test keyboard focus on every control</button>
      <ProbeList results={probe} />
      <p aria-live="polite" style={{fontWeight: 600}}>{notes[mode]}</p>
      <p style={{fontSize: "0.85rem", opacity: 0.8}}>
        The test calls focus() on each control and checks whether the browser accepted it, which is the same thing the Tab key does. Nothing is stored or sent.
      </p>
    </div>
  );
}

export function MenuLab() {
  const menuRef = useRef(null);
  const [open, setOpen] = useState(false);
  const [useInert, setUseInert] = useState(false);
  const [probe, setProbe] = useState(null);
  useEffect(function () {
    const el = menuRef.current;
    if (el && "inert" in el) el.inert = useInert && !open;
    setProbe(null);
  }, [open, useInert]);
  return (
    <div style={{padding: "1rem"}}>
      <label style={{display: "block", margin: "0 0 0.5rem"}}>
        <input type="checkbox" checked={useInert} onChange={function (e) { setUseInert(e.target.checked); }} /> Set inert on the menu while it is closed
      </label>
      <button type="button" aria-expanded={open} onClick={function () { setOpen(!open); }}>{open ? "Close menu" : "Open menu"}</button>

    </div>

Sometimes part of a page should be **out of reach**: a menu that has slid off screen, a section that is saving, the page behind a pop-up. Before `inert`, you switched these off by hand, with `tabindex="-1"` on every link, `aria-hidden` on containers, `pointer-events: none` in CSS, and a lot of luck. The **`inert` attribute** does the whole job with one word.

## The Big Idea

Putting `inert` on an element says: "**this part of the page is not available right now.**" The element and everything inside it stop taking part in the page, for **every** kind of visitor: mouse, touch, keyboard, and screen reader.

## What inert Does

For an inert element and **all of its descendants**:

| Part of the experience | What happens |
| :-- | :-- |
| **Pointer** | Clicks and taps do nothing, because the element is skipped when the browser works out what was pressed |
| **Keyboard** | Nothing inside can take focus, so Tab skips it |
| **Screen readers** | The content is removed from the accessibility tree, as if it were hidden |
| **Text selection** | Text cannot be selected |
| **Find in page** | Text inside is generally not found |
| **Looks** | **Nothing changes.** It still shows, exactly as before |

That last row matters: `inert` is about **behaviour**, not appearance. If a section is switched off, show that with your own CSS, such as lower opacity, and say so in text.

Open the **Focus Lab** and try the buttons in each mode. In the plain mode, every control accepts focus. In the `inert` mode, every control refuses it, and clicking the Buy button does nothing.

## inert, disabled, hidden, and aria-hidden

These four are easy to mix up:

| Tool | Where it works | Looks | Focus and clicks | Screen reader | Form data |
| :-- | :-- | :-- | :-- | :-- | :-- |
| `inert` | Any element, with its whole subtree | Unchanged | Blocked | Hidden | Not what it is for |
| `disabled` | Form controls (and `fieldset`) | Usually greyed out | Blocked | Announced as unavailable | The control is **left out** of the submission |
| `hidden` | Any element | Not displayed, no space used | Not reachable | Hidden | Not rendered |
| `aria-hidden="true"` | Any element | Unchanged | **Not blocked** | Hidden | Unchanged |

### The aria-hidden trap

`aria-hidden="true"` only tells **screen readers** to ignore something. A keyboard user can still Tab into it. So a link inside an `aria-hidden` area can take focus, while a screen reader says nothing about it. That is a real accessibility failure.

Switch the Focus Lab to `aria-hidden="true" only` and run the focus test: every control still accepts focus. `inert` fixes both problems at once. Keep `aria-hidden` for things that **cannot** be focused, such as a purely decorative icon.

:::info
`inert` is about interaction, not form data. If you need a control left out of a submission, use `disabled` on that control. Do not depend on `inert` for that.
:::

## Pattern 1: Off-Canvas Menus

A menu that slides off screen with CSS is **invisible, but not gone.** Its links are still in the tab order, so a keyboard user tabs into links they cannot see, and a screen reader reads a menu that looks closed.

The fix: set `inert` while the menu is closed, and remove it when the menu opens, as in the code above. Open the **Off-Canvas Menu Lab**, leave the box unticked, and run the focus test on the closed menu. Then tick the box and run it again.

## Pattern 2: A Section That Is Saving

When a large part of a form is saving, you want it frozen so no one edits it half-way. Instead of disabling every field, make the whole section inert and tell people what is going on:

```html
<p id="status" role="status"></p>
<fieldset id="details">...many fields...</fieldset>
```

```js
details.inert = true;
status.textContent = "Saving your changes...";
// ...when done...
details.inert = false;
status.textContent = "Saved.";
```

Notice that the status message sits **outside** the inert area. An inert area is hidden from screen readers, so anything you want announced must live somewhere else.

## Pattern 3: Slides and Hidden Panels

In a carousel, the slides that are off screen are still in the page. Make every slide except the current one `inert`, so Tab does not wander into slides nobody can see. The same idea fits stacked panels, such as the earlier steps of a wizard that you keep in the page but not in view.

If you do not need to keep an area in the layout at all, `hidden` is simpler.

## Pattern 4: Behind a Custom Overlay

A native `dialog` opened with `showModal()` already makes the rest of the page behave as if it were inert, as the [Dialog lesson](./dialog) explains. If you cannot use a `dialog` and build your own overlay, set `inert` on the page regions behind it (not on the overlay itself), and remove it when the overlay closes. Use the native dialog whenever you can, because it also handles focus and the Escape key.

## Setting inert From JavaScript

`inert` is a boolean attribute, and it has a matching JavaScript property that stays in sync:

```js
panel.inert = true;     // adds the attribute
panel.inert = false;    // removes it
console.log(panel.inert);
```

A few rules to remember:

- **It is inherited and cannot be undone below.** If a parent is inert, a child cannot opt out, even with `inert="false"`. Remove it from the parent.
- **Move focus first.** If the element that currently has focus becomes inert, focus is lost and falls back to the page. Move focus somewhere sensible before you switch a region off, for example to the control that opened it.
- **Explain the state.** Use text, a live region outside the inert area, and your own styling.

## Support and Fallbacks

`inert` works in current versions of the major browsers, but older versions ignore it. Check its Baseline label on MDN. In a browser without support:

- the area simply **stays interactive**,
- and nothing breaks, but you lose the protection.

For a low-stakes case, such as an off-screen menu, that is acceptable, and you can detect support if you want a backup:

```js
if (!("inert" in HTMLElement.prototype)) {
  // optional: add tabindex="-1" to focusable children while the area is off
}
```

For high-stakes cases, such as blocking the page behind a pop-up, prefer the native `dialog`, or a maintained polyfill.

## Accessibility Notes

- **Do not make visible, important content inert** without a clear reason and a visible explanation. People who use screen readers cannot reach it, and nobody can select or copy its text.
- **Tell visitors why.** A greyed-out area with no explanation is confusing. Add a visible message such as "Saving...", in a live region **outside** the inert area.
- **Manage focus on purpose** when you add or remove `inert`, as above.
- **Do not use `inert` to hide content** from screen readers when it should be hidden from everyone. Use `hidden`.
- **Test it:** press Tab through the page, click the area, and look at the browser's accessibility panel, if it has one, to confirm the area is gone from the tree.

## Checking Your Setup

1. Press **Tab** repeatedly. Does focus ever land on something you cannot see?
2. **Click** inside the area you switched off. Does anything respond?
3. In DevTools, open the **Accessibility** panel and check that the inert area is not listed.
4. Switch the state on and off, and watch where **focus** goes.
5. Test with a **screen reader** if you can, and listen for your status message.
6. Look for `aria-hidden` on containers that hold links or buttons. They are the likely bugs.
7. Test in a **second browser**, and think about what happens without support.

## Best Practices

1. **Use `inert`** for areas that are temporarily unavailable.
2. **Prefer it to `aria-hidden`** on anything that can take focus.
3. **Use `disabled`** for individual form controls, and `hidden` to remove things entirely.
4. **Show the state** with CSS and text.
5. **Put announcements outside** the inert area.
6. **Move focus** before inerting the region that has it.
7. **Use a native `dialog`** for modal pop-ups.
8. **Plan for browsers without support.**
9. **Keep the area small and temporary.**
10. **Test with the keyboard and a screen reader.**

## Security and Privacy Notes

- **`inert` is not security.** It is a client-side setting that any visitor can remove in DevTools. Never use it to protect information or actions. The server must decide who may do what.
- **Inert content is still in the page.** It is in the source, and it may still be sent in a form. Do not hide private data in an inert area.
- **Do not rely on it to prevent double submissions** or double payments. Disable buttons, and make the server ignore repeats. See [Secure Forms](/tutorials/html/security/secure-forms).
- **Inert does not stop scripts.** A script can still read or change an inert area.

## Global Audience Notes

- **Keyboard and screen reader use is common** among many visitors, including people who cannot use a mouse. Hidden-but-focusable content hurts them the most.
- **Explain states in plain, translatable text.** "Saving..." and "Unavailable" are easier to translate than icons alone.
- **Slow connections make saving states long.** A clear message and a way to tell when it ends are especially helpful.
- **Older and low-cost devices** may lack support. Keep the page usable without it.

## Common Mistakes to Avoid

1. **Hiding a menu with CSS only,** so its links still take focus.
2. **Using `aria-hidden`** on areas with links or buttons.
3. **Making important content inert** with no explanation.
4. **Putting the status message inside** the inert area.
5. **Forgetting to move focus** before switching the area off.
6. **Expecting `inert` to change how the area looks.**
7. **Using `inert` instead of `disabled`** to keep a field out of a submission.
8. **Trying to un-inert a child** of an inert parent.
9. **Building a custom pop-up** when a native `dialog` would do.
10. **Treating `inert` as protection.**

## Key Takeaways

- **`inert` switches off an area for pointer, keyboard, and screen reader,** with one attribute.
- **It changes behaviour, not looks.** Show the state yourself.
- **Prefer it to `aria-hidden`** whenever the content can take focus.
- **Use `disabled` for single controls and `hidden` to remove content.**
- **Great for off-canvas menus, saving states, and carousels.**
- **Move focus first, and announce outside the area.**
- **Never treat it as security.**
